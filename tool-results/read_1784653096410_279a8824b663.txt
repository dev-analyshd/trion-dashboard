   733→    const HOSTILE: u8    = 3_u8;

   737→    const SAFE_GC_MIN: u64       = 4000_u64;
   738→    const CAUTION_ANIMA_MIN: u64 = 2500_u64;
   739→    const CAUTION_GC_MIN: u64    = 1500_u64;
   740→
   741→    // Trajectory alert levels
   742→    const ALERT_WARN: u8         = 1_u8;
   743→    const ALERT_MANIPULATION: u8 = 2_u8;
   744→
   745→    #[storage]
   746→    struct Storage {
   747→        owner: ContractAddress,
   748→        oracle: ContractAddress,

   789→    #[generate_trait]
   790→    impl InternalImpl of InternalTrait {
   791→        fn compute_tier(
   792→            anima_score: u64,
   793→            trajectory_alert: u8,
   794→            genesis_confidence: u64,
   795→            update_count: u64,
   796→        ) -> u8 {
   797→            // HOSTILE: active TRION manipulation signal
   798→            if trajectory_alert == ALERT_MANIPULATION {
   799→                return HOSTILE;
   800→            }
   801→            // HIGH_RISK: no behavioral history on-chain
   802→            if update_count == 0 {
   803→                return HIGH_RISK;
   804→            }
   805→            // HIGH_RISK: extremely weak scores
   806→            if anima_score < CAUTION_ANIMA_MIN && genesis_confidence < CAUTION_GC_MIN {
   807→                return HIGH_RISK;
   808→            }
   809→            // CAUTION: partial history or warning
   810→            if trajectory_alert == ALERT_WARN
   811→                || anima_score < SAFE_ANIMA_MIN
   812→                || genesis_confidence < SAFE_GC_MIN {
   813→                return CAUTION;
   814→            }
   815→            SAFE
   816→        }
   817→    }

   820→    impl BTCFiGuardImpl of IBTCFiGuard<ContractState> {
   821→        fn assess_risk(self: @ContractState, beo_id: felt252) -> u8 {
   822→            let oracle = ITRIONOracleDispatcher { contract_address: self.oracle.read() };
   823→            let score = oracle.get_score(beo_id);
   824→            InternalImpl::compute_tier(
   825→                score.anima_score,
   826→                score.trajectory_alert,
   827→                score.genesis_confidence,
   828→                score.update_count,
   829→            )
   830→        }
   831→
   832→        fn batch_assess(self: @ContractState, beo_ids: Span<felt252>) -> Array<u8> {
   833→            let oracle = ITRIONOracleDispatcher { contract_address: self.oracle.read() };
   834→            let mut results: Array<u8> = ArrayTrait::new();
   835→            let mut i: usize = 0;
   836→            loop {
   837→                if i >= beo_ids.len() { break; }
   838→                let beo_id = *beo_ids.at(i);
   839→                let score = oracle.get_score(beo_id);
   840→                results.append(
   841→                    InternalImpl::compute_tier(
   842→                        score.anima_score,
   843→                        score.trajectory_alert,
   844→                        score.genesis_confidence,
   845→                        score.update_count,
   846→                    )
   847→                );
   848→                i += 1;
   849→            };
   850→            results
   851→        }
   852→
   853→        fn score_to_tier(
   854→            self: @ContractState,
   855→            anima_score: u64,
   856→            trajectory_alert: u8,
   857→            genesis_confidence: u64,
   858→        ) -> u8 {
   859→            InternalImpl::compute_tier(anima_score, trajectory_alert, genesis_confidence, 1)
   860→        }

   862→        fn get_safe_threshold(self: @ContractState) -> u8 {
   863→            self.safe_threshold.read()
   864→        }
   865→

   867→            let caller = get_caller_address();
   868→            assert(caller == self.owner.read(), 'BTCFi: unauthorized');
   869→            assert(threshold <= 2_u8, 'BTCFi: threshold must be 0-2');
   870→            let old = self.safe_threshold.read();
   871→            self.safe_threshold.write(threshold);
   872→            self.emit(ThresholdUpdated { old_threshold: old, new_threshold: threshold });
   873→        }
   874→
   875→        fn get_oracle(self: @ContractState) -> ContractAddress {
   876→            self.oracle.read()
   877→        }
   878→
   879→        fn set_oracle(ref self: ContractState, oracle: ContractAddress) {
   880→            let caller = get_caller_address();
   881→            assert(caller == self.owner.read(), 'BTCFi: unauthorized');
   882→            let old = self.oracle.read();
   883→            self.oracle.write(oracle);
   884→            self.emit(OracleUpdated { old_oracle: old, new_oracle: oracle });
   885→        }
   886→
   887→        fn get_owner(self: @ContractState) -> ContractAddress {
   888→            self.owner.read()
   889→        }
   890→
   891→        fn transfer_ownership(ref self: ContractState, new_owner: ContractAddress) {
   892→            let caller = get_caller_address();
   893→            assert(caller == self.owner.read(), 'BTCFi: unauthorized');
   894→            let old = self.owner.read();
   895→            self.owner.write(new_owner);
   896→            self.emit(OwnershipTransferred { previous_owner: old, new_owner });
   897→        }
   898→    }

   951→function encodeTrajectoryAlert(alert: string): number {
   952→  if (alert === 'MANIPULATION_ALERT') return TRAJECTORY_ALERT.MANIPULATION;
   953→  if (alert === 'WARN')               return TRAJECTORY_ALERT.WARN;
   954→  return TRAJECTORY_ALERT.CLEAR;

   957→function encodeDormancyType(dt: string): bigint {
   958→  const safe = (dt ?? 'UNKNOWN').slice(0, 31);
   959→  return BigInt(shortString.encodeShortString(safe));
   960→}

   962→function scaleScore(val: number | null | undefined, scale = 10000): bigint {
   963→  if (val === null || val === undefined || isNaN(val)) return 0n;
   964→  return BigInt(Math.round(Math.min(1, Math.max(0, val)) * scale));
   965→}

   967→async function fetchEntityScores(entityId: string) {
   968→  const faissBase = STARKNET_CONFIG.trion.faissBaseUrl;

   971→    axios.get<FAISSTrajectory>(`${faissBase}/api/v1/trajectory_anomaly/${entityId}`, { timeout: 8000 }),
   972→    axios.get<FAISSResurrection>(`${faissBase}/api/v1/resurrection_status/${entityId}`, { timeout: 8000 }),
   973→  ]);
   974→
   975→  const traj = trajRes.status === 'fulfilled' ? trajRes.value.data : null;
   976→  const res  = resRes.status  === 'fulfilled' ? resRes.value.data  : null;
   977→
   978→  return { traj, res };
   979→}

   991→  const archetypeId       = BigInt(Math.max(0, Math.min(63, traj?.archetype_id ?? 0)));
   992→  const akashicDepth      = scaleScore(res?.delta_score ?? 0, 1);
   993→  const isResurrection    = res?.is_resurrection ?? false;
   994→  const dormancyType      = encodeDormancyType(res?.dormancy_type ?? 'UNKNOWN');
   995→
   996→  const beoBigInt = BigInt('0x' + beoId.slice(0, 31));

   998→  console.log(`  Pushing score for ${entityId}:`);

  1129→export async function getWorkingProvider(): Promise<RpcProvider> {
  1130→  // Probe each endpoint with BOTH a chainId check AND a real getBlockWithTxs call.
  1131→  // Some providers (e.g. drpc.org) advertise the right chainId but return
  1132→  // -32601 "method does not exist" for starknet_getBlockWithTxs — that breaks
  1133→  // the indexer hot loop. We must reject those endpoints up-front.
  1134→  for (const url of STARKNET_CONFIG.rpcEndpoints) {
  1135→    try {
  1136→      const provider = new RpcProvider({ nodeUrl: url });
  1137→      const chainId = await provider.getChainId();
  1138→      if (chainId !== STARKNET_CONFIG.chainIdHex) continue;
  1139→      // Probe getBlockWithTxs against latest block — must succeed.
  1140→      const latest = await provider.getBlockNumber();
  1141→      await provider.getBlockWithTxs(latest);
  1142→      console.log(`✓ Connected to Starknet Mainnet via ${url}`);
  1143→      console.log(`  Chain ID: ${chainId} (SN_MAIN), latest block: ${latest}`);
  1144→      return provider;
  1145→    } catch (e: any) {
  1146→      const msg = e?.message ?? String(e);
  1147→      console.log(`  · RPC ${url} rejected: ${msg.slice(0, 120)}`);
  1148→    }
  1149→  }
  1150→  throw new Error('All Starknet Mainnet RPC endpoints failed (none support getBlockWithTxs)');
  1151→}
