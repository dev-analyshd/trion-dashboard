     1→   501→        return 0.0
     2→   502→    return h / math.log2(n)
     3→   503→
     4→   504→
     5→   505→def features_from_block(block: dict) -> tuple[list[float], dict]:
     6→   506→    """Compute the 9 Φ-features + per-vector data for a Solana block."""
     7→   507→    txs = block.get("transactions") or []
     8→   508→    fee_counter = Counter()
     9→   509→    cu_counter = Counter()        # log10-bucketed CU (used for f5)
    10→   510→    cu_linear_counter = Counter() # linear-bucketed CU (used for f8 — distinct from f5)
    11→   511→    program_counter = Counter()
    12→   512→    account_counter = Counter()
    13→   513→    instr_count_counter = Counter()
    14→   514→    err_counter = Counter()
    15→   515→    acct_per_tx_counter = Counter()   # for f7: accounts-per-tx complexity entropy
    16→   516→    joint_fee_cu_counter = Counter()  # for f9: joint fee × CU correlation entropy
    17→   517→
    18→   518→    total_lamports = 0
    19→   519→    successful = 0
    20→   520→    failed = 0
    21→   521→
    22→   522→    for entry in txs:
    23→   523→        meta = entry.get("meta") or {}
    24→   524→        tx = entry.get("transaction") or {}
    25→   525→        msg = (tx.get("message") if isinstance(tx, dict) else {}) or {}
    26→   526→
    27→   527→        fee = int(meta.get("fee") or 0)
    28→   528→        cu = int(meta.get("computeUnitsConsumed") or 0)
    29→   529→        err = meta.get("err")
    30→   530→
    31→   531→        # Bin lamport fees and CU into log buckets for entropy stability
    32→   532→        fee_bucket = 0 if fee == 0 else int(math.floor(math.log10(fee + 1)))
    33→   533→        cu_bucket = 0 if cu == 0 else int(math.floor(math.log10(cu + 1)))
    34→   534→        fee_counter[fee_bucket] += 1
    35→   535→        cu_counter[cu_bucket] += 1
    36→   536→        # Linear CU bucket for f8 (distinct from log10 cu_bucket used in f5)
    37→   537→        cu_lin_bucket = (0 if cu == 0 else
    38→   538→                         1 if cu < 50_000 else
    39→   539→                         2 if cu < 200_000 else
    40→   540→                         3 if cu < 500_000 else
    41→   541→                         4 if cu < 1_000_000 else 5)
    42→   542→        cu_linear_counter[cu_lin_bucket] += 1
    43→   543→
    44→   544→        if err is None:
    45→   545→            successful += 1
    46→   546→            err_counter["ok"] += 1
    47→   547→        else:
    48→   548→            failed += 1
    49→   549→            err_counter["err"] += 1
    50→   550→
    51→   551→        accts = msg.get("accountKeys") or []
    52→   552→        for a in accts:
    53→   553→            # accountKeys may be strings or {"pubkey": ...} objects
    54→   554→            key = a if isinstance(a, str) else a.get("pubkey") if isinstance(a, dict) else None
    55→   555→            if key:
    56→   556→                account_counter[key] += 1
    57→   557→
    58→   558→        # f7 tracking: bin accounts-per-tx into log2 buckets (tx complexity diversity)
    59→   559→        n_accts = len(accts)
    60→   560→        acct_bin = 0 if n_accts == 0 else min(5, int(math.floor(math.log2(n_accts + 1))))
    61→   561→        acct_per_tx_counter[acct_bin] += 1
    62→   562→
    63→   563→        # f9 tracking: joint (fee_bucket, cu_bucket) for correlation entropy
    64→   564→        joint_fee_cu_counter[(fee_bucket, cu_bucket)] += 1
    65→   565→
    66→   566→        instrs = msg.get("instructions") or []
    67→   567→        instr_count_counter[len(instrs)] += 1
    68→   568→        for ix in instrs:
    69→   569→            pid_idx = ix.get("programIdIndex")
    70→   570→            if isinstance(pid_idx, int) and 0 <= pid_idx < len(accts):
    71→   571→                pid = accts[pid_idx] if isinstance(accts[pid_idx], str) else (accts[pid_idx].get("pubkey") if isinstance(accts[pid_idx], dict) else None)
    72→   572→                if pid:
    73→   573→                    program_counter[pid] += 1
    74→   574→
    75→   575→        total_lamports += fee
    76→   576→
    77→   577→    # f1..f9 per L1.1 (Solana-mapped):
    78→   578→    # f1 fee entropy, f2 account diversity, f3 tx-success ratio entropy,
    79→   579→    # f4 program diversity, f5 cu entropy, f6 instr-count entropy,
    80→   580→    # f7 program concentration (1-Gini-style proxy via top-k share),
    81→   581→    # f8 cu/fee correlation proxy, f9 reserved (set to overall block entropy).
    82→   582→    f1 = shannon(fee_counter)
    83→   583→    f2 = shannon(account_counter)
    84→   584→    f3 = shannon(err_counter)
    85→   585→    f4 = shannon(program_counter)
    86→   586→    f5 = shannon(cu_counter)
    87→   587→    f6 = shannon(instr_count_counter)
    88→   588→    # f7: Shannon entropy of accounts-per-tx distribution (tx complexity diversity)
    89→   589→    #     H(log2-binned account counts per tx) — distinct from f2 (unique accounts)
    90→   590→    f7 = shannon(acct_per_tx_counter)
    91→   591→    # f8: Shannon entropy of CU linear-bucket distribution (distinct from f5)
    92→   592→    #     f5 uses log10 buckets (scale-invariant); f8 uses linear buckets (scale-sensitive)
    93→   593→    #     This captures whether CU distribution is skewed toward specific cost tiers
    94→   594→    f8 = shannon(cu_linear_counter)
    95→   595→    # f9: Shannon entropy of joint (fee_bucket × cu_bucket) distribution
    96→   596→    #     Captures fee-compute correlation patterns — independent signal from f1/f5
    97→   597→    f9 = shannon(joint_fee_cu_counter)
    98→   598→
    99→   599→    feats = [f1, f2, f3, f4, f5, f6, f7, f8, f9]
   100→   600→
   101→   601→    stats = {
   102→   602→        "tx_count": len(txs),
   103→   603→        "successful": successful,
   104→   604→        "failed": failed,
   105→   605→        "total_fee_lamports": total_lamports,
   106→   606→        "unique_programs": len(program_counter),
   107→   607→        "unique_accounts": len(account_counter),
   108→   608→        "phi": sum(feats) / 9.0,
   109→   609→    }
   110→   610→    return feats, stats
   111→   611→
   112→   612→
   113→   613→def vector_from_features(feats: list[float], slot: int) -> list[float]:
   114→   614→    """Expand the 9 features into a deterministic 128-dim behavioural vector.
   115→   615→    Bands 0-15 = f1, 16-31 = f2, ... 128-143 = f9 (last 17 reserved for slot
   116→   616→    salt to prevent collapse to a single FAISS centroid).
   117→   617→    """
   118→   618→    out: list[float] = []
   119→   619→    band = DIM // 9
   120→   620→    for v in feats:
   121→   621→        out.extend([float(v)] * band)
   122→   622→    # pad to DIM with a deterministic slot-derived salt
   123→   623→    seed = hashlib.sha256(f"{slot}".encode()).digest()
   124→   624→    while len(out) < DIM:
   125→   625→        b = seed[(len(out) - DIM) % 32]
   126→   626→        out.append((b / 255.0) - 0.5)
   127→   627→    return out[:DIM]
   128→   628→

   130→   630→def push_block(slot: int, feats: list[float], stats: dict) -> None:
   131→   631→    entity_id = f"solana:slot:{slot}"

   133→   633→    payload = {
   134→   634→        "vectors": [{
   135→   635→            "entity_id":   entity_id,
   136→   636→            "vector":      vector,
   137→   637→            "magnitude":   max(1.0, math.log10(stats["total_fee_lamports"] + 1)),
   138→   638→            "entropy":     stats["phi"],
   139→   639→            "block_num":   slot,
   140→   640→            "chain_id":    SOLANA_CHAIN_ID,

   142→   642→            "vm_type":     "SVM",
   143→   643→        }],

   147→   647→        "entity_id":      entity_id,

   162→   662→    # Circuit breaker: when FAISS has been failing, back off so we don't
   163→   663→    # pile more requests into its queue (which is the main cause of
   164→   664→    # cascading 15s timeouts). Returns True on success, False on failure.
   165→   665→    global _faiss_failures, _faiss_skip_until
   166→   666→    now_ts = time.monotonic()

   168→   668→        return False
   169→   669→    try:

   171→   671→        # for 15s per slot. If FAISS can't respond in 4s it's saturated and
   172→   672→        # retrying will only make it worse — let the circuit breaker kick in.
   173→   673→        r = session.post(url, data=json.dumps(payload), timeout=4)
   174→   674→        if r.status_code != 200:
   175→   675→            print(f"[svm][slot={slot}] FAISS POST {r.status_code}: {r.text[:200]}", file=sys.stderr)
   176→   676→            _faiss_failures += 1
   177→   677→        else:

   180→   680→                  f"fail={stats['failed']:>3} prog={stats['unique_programs']:>3} "
   181→   681→                  f"acct={stats['unique_accounts']:>4} Φ={stats['phi']:.4f} "
   182→   682→                  f"→ added={j.get('added',0)} idx_total={j.get('indexed_vectors','?')}")

   185→   685→    except requests.RequestException as e:
   186→   686→        print(f"[svm][slot={slot}] FAISS POST failed: {e}", file=sys.stderr)
   187→   687→        _faiss_failures += 1
   188→   688→    # On 3+ consecutive failures: open circuit for 30s, giving FAISS room
   189→   689→    # to drain its queue and serve oracle reads (mental/spiritual/etc).
   190→   690→    if _faiss_failures >= 3:
   191→   691→        _faiss_skip_until = time.monotonic() + 30.0
   192→   692→        print(f"[svm] FAISS circuit OPEN for 30s (consecutive failures={_faiss_failures})", file=sys.stderr)
   193→   693→        _faiss_failures = 0
   194→   694→    return False

   202→   702→def main() -> int:
   203→   703→    print("===============================================================")
   204→   704→    print(" TRION L0-SVM: SOLANA BEHAVIORAL INDEXER")
   205→   705→    print("===============================================================")
   206→   706→    print(f" RPC          : {SOLANA_RPC_URL}")
   207→   707→    print(f" Chain id     : {SOLANA_CHAIN_ID}  ({SOLANA_LABEL})")
   208→   708→    print(f" FAISS target : {FAISS_SERVICE_URL}")
   209→   709→    print(f" Poll         : {POLL_SLEEP_MS}ms")
   210→   710→    print(" Streaming live Solana slots into FAISS. Press Ctrl+C to stop.")
   211→   711→    print()
   212→   712→
   213→   713→    # Public Solana RPCs typically only serve `getBlock` for slots that are
   214→   714→    # already FINALIZED. To stay reliably inside that window we keep our
   215→   715→    # cursor a fixed offset behind the finalized tip. SAFE_LAG = 40 slots
   216→   716→    # ≈ 16 s — well past the 32-block finality window.
   217→   717→    SAFE_LAG = 40
   218→   718→    last_slot = 0
   219→   719→    consecutive_failures = 0
   220→   720→
   221→   721→    while True:
   222→   722→        try:
   223→   723→            current = rpc("getSlot", [{"commitment": "finalized"}])
   224→   724→            if not isinstance(current, int):
   225→   725→                time.sleep(POLL_SLEEP_MS / 1000.0)
   226→   726→                continue
   227→   727→
   228→   728→            if last_slot == 0:
   229→   729→                last_slot = current - SAFE_LAG
   230→   730→
   231→   731→            # Stay at least SAFE_LAG behind the finalized tip
   232→   732→            tip = current - SAFE_LAG
   233→   733→            if tip <= last_slot:
   234→   734→                time.sleep(POLL_SLEEP_MS / 1000.0)
   235→   735→                continue
   236→   736→
   237→   737→            target = last_slot + 1
   238→   738→            try:
   239→   739→                block = rpc("getBlock", [target, {
   240→   740→                    "encoding": "json",
   241→   741→                    "transactionDetails": "full",
   242→   742→                    "maxSupportedTransactionVersion": 0,
   243→   743→                    "rewards": False,
   244→   744→                    "commitment": "finalized",
   245→   745→                }])
   246→   746→                consecutive_failures = 0
   247→   747→            except RuntimeError as e:
   248→   748→                msg = str(e)
   249→   749→                # Skipped slot (-32007 / -32009) — advance silently
   250→   750→                if "-32007" in msg or "-32009" in msg or "skipped" in msg.lower():
   251→   751→                    last_slot = target
   252→   752→                    continue
   253→   753→                # Slot data missing (-32004) — typically the public node hasn't
   254→   754→                # caught up. Skip ahead a bit and retry rather than spinning.
   255→   755→                if "-32004" in msg or "not available" in msg.lower() or "older than" in msg.lower():
   256→   756→                    consecutive_failures += 1
   257→   757→                    # Jump forward by min(consecutive_failures * 5, 100) slots,
   258→   758→                    # but never past the safe tip.
   259→   759→                    jump = min(consecutive_failures * 5, 100)
   260→   760→                    last_slot = min(last_slot + jump, tip)
   261→   761→                    if consecutive_failures % 20 == 1:
   262→   762→                        print(f"[svm] block missing for slot {target}, advancing +{jump} (cursor={last_slot}, tip={tip})", file=sys.stderr)
   263→   763→                    time.sleep(0.3)
   264→   764→                    continue
   265→   765→                raise
   266→   766→
   267→   767→            if not block:
   268→   768→                last_slot = target
   269→   769→                continue
   270→   770→
   271→   771→            feats, stats = features_from_block(block)
   272→   772→            push_block(target, feats, stats)
   273→   773→            last_slot = target
   274→   774→            consecutive_failures = 0
   275→   775→
   276→   776→        except KeyboardInterrupt:
   277→   777→            print("\n[svm] shutdown requested")
   278→   778→            return 0
   279→   779→        except requests.HTTPError as e:
   280→   780→            # 429 = rate limited. Public Solana RPCs throttle aggressively;
   281→   781→            # back off for a longer window and resume from current confirmed slot.
   282→   782→            status = getattr(e.response, "status_code", None)
   283→   783→            if status == 429:
   284→   784→                print(f"[svm] rate-limited (429), sleeping 15s", file=sys.stderr)
   285→   785→                time.sleep(15.0)
   286→   786→                last_slot = 0   # re-bootstrap from current confirmed slot
   287→   787→            else:
   288→   788→                print(f"[svm] HTTP {status}: {e}", file=sys.stderr)
   289→   789→                time.sleep(3.0)
   290→   790→        except Exception as e:
   291→   791→            msg = str(e)
   292→   792→            if "429" in msg or "Too Many Requests" in msg:
   293→   793→                print(f"[svm] rate-limited (429), sleeping 15s", file=sys.stderr)
   294→   794→                time.sleep(15.0)
   295→   795→                last_slot = 0
   296→   796→            else:
   297→   797→                print(f"[svm] loop error: {e}", file=sys.stderr)
   298→   798→                time.sleep(2.0)
   299→   799→
   300→   800→
   301→   801→if __name__ == "__main__":
   302→   802→    sys.exit(main())
