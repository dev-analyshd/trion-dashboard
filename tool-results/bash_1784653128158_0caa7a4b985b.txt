 * TRION SUI — Real Transaction Executor + BTCP Oracle Anchor
 *
 * Sends 5 real SUI transactions on Sui Mainnet, ingests behavioral vectors into FAISS,
 * and anchors oracle proofs on-chain.
 *
 * Usage:  SUI_PRIVATE_KEY=suipri... tsx execute.ts

import fs from "fs";
import { createRequire } from "module";
const _require = createRequire(import.meta.url);

const FAISS_URL = process.env.FAISS_URL ?? "http://127.0.0.1:8000";
const CHAIN_ID  = 101;
const VM_TYPE   = "SUI";
const SUI_RPC   = process.env.SUI_RPC ?? "https://fullnode.mainnet.sui.io/";

function sleep(ms: number) { return new Promise(r => setTimeout(r, ms)); }

function shannonEntropy(values: number[]): number {
  const total = values.reduce((a, b) => a + b, 0);
  if (total === 0) return 0;
  return -values.filter(v => v > 0).map(v => { const p = v / total; return p * Math.log2(p); }).reduce((a, b) => a + b, 0);
}

function makeVector(checkpoint: number, txCount: number, gasUsed: number): number[] {
  const v = new Array(128).fill(0);
  const e = shannonEntropy([txCount + 1, gasUsed + 1, checkpoint % 100 + 1]);
  v[0] = Math.min(1, txCount / 50);
  v[1] = Math.min(1, gasUsed / 1e9);
  v[2] = e / 3.0;
  v[3] = (checkpoint % 1000) / 1000;
  for (let i = 4; i < 128; i++) v[i] = Math.abs(Math.sin(checkpoint * (i + 1))) * 0.1;
  return v;
}

async function ingestToFaiss(entityId: string, vector: number[], phi: number): Promise<any> {
  try {
    const res = await fetch(`${FAISS_URL}/index/add_batch`, {
      method: "POST", headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ vectors: [{ entity_id: entityId, vector, magnitude: phi, entropy: vector[2], chain_id: CHAIN_ID, vm_type: VM_TYPE }] }),
      signal: AbortSignal.timeout(10000),
    });
    return await res.json();
  } catch (e: any) { return { error: e.message }; }
}

async function suiRpc(method: string, params: any[]): Promise<any> {
  const res = await fetch(SUI_RPC, {
    method: "POST", headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ jsonrpc: "2.0", id: 1, method, params }),
    signal: AbortSignal.timeout(15000),
  });
  const d = await res.json() as any;
  if (d.error) throw new Error(`SUI RPC: ${JSON.stringify(d.error)}`);
  return d.result;
}

async function getBalance(address: string): Promise<bigint> {
  try {
    const r = await suiRpc("suix_getBalance", [address, "0x2::sui::SUI"]);
    return BigInt(r.totalBalance || "0");
  } catch { return 0n; }
}

async function getLatestCheckpoint(): Promise<number> {
  try {
    const r = await suiRpc("sui_getLatestCheckpointSequenceNumber", []);
    return parseInt(r);
  } catch { return 0; }
}

  }

  console.log("\n  Results → /tmp/sui_execution_results.json");
}

async function probeRpc(url: string, timeoutMs = 8000): Promise<boolean> {
  try {
    const ctrl = new AbortController();
    const timer = setTimeout(() => ctrl.abort(), timeoutMs);
    const res = await fetch(url, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ jsonrpc: "2.0", id: 1, method: "getSlot", params: [] }),
      signal: ctrl.signal as any,
    });
    clearTimeout(timer);
    if (!res.ok) return false;
    const data: any = await res.json();
    return typeof data.result === "number";
  } catch {
    return false;
  }
}

async function findWorkingRpc(): Promise<string> {
  console.log("  Probing Solana mainnet RPC endpoints...");
  for (const url of MAINNET_RPCS) {
    process.stdout.write(`    ${url} ... `);
    const ok = await probeRpc(url);
    console.log(ok ? "✓ LIVE" : "✗ down");
    if (ok) return url;
  }
  throw new Error(
    "All Solana Mainnet RPC endpoints are unreachable right now. " +
    "Check network connectivity or set SOLANA_RPC to a dedicated endpoint."
  );
}

#!/usr/bin/env python3
"""
TRION L0-SVM — Solana Behavioral Indexer
=========================================
Polls Solana mainnet RPC slot-by-slot, computes whitepaper L1.1 Φ-features
(Shannon entropies of program-id, fee, compute-unit, account-touch and
instruction-type distributions) for each block, and POSTs a 128-dim behavioral
vector batch to the FAISS service with vm_type="SVM".

Mirrors the contract followed by trion-l0/src/main.rs for EVM chains, but uses
Solana JSON-RPC (getSlot / getBlock) instead of eth_blockNumber / eth_getBlock.

Environment:
  SOLANA_RPC_URL    Solana JSON-RPC endpoint  (default: https://api.mainnet-beta.solana.com)
  SOLANA_CHAIN_ID   Numeric chain id          (default: 101  — mainnet-beta)
  SOLANA_LABEL      Human label               (default: "SOLANA_MAINNET")
  FAISS_SERVICE_URL FAISS ingest target       (default: http://127.0.0.1:8000)
  POLL_SLEEP_MS     Slot poll interval        (default: 800ms — Solana slot ≈400ms)
"""
from __future__ import annotations

import hashlib
import json
import math
import os
import sys
import time
from collections import Counter
from typing import Any

import requests

SOLANA_RPC_URL    = os.environ.get("SOLANA_RPC_URL", "https://api.mainnet-beta.solana.com")
SOLANA_CHAIN_ID   = int(os.environ.get("SOLANA_CHAIN_ID", "101"))
SOLANA_LABEL      = os.environ.get("SOLANA_LABEL", "SOLANA_MAINNET")
FAISS_SERVICE_URL = os.environ.get("FAISS_SERVICE_URL", "http://127.0.0.1:8000").rstrip("/")
POLL_SLEEP_MS     = int(os.environ.get("POLL_SLEEP_MS", "800"))

DIM = 128

session = requests.Session()
session.headers.update({"Content-Type": "application/json"})


def rpc(method: str, params: list[Any]) -> Any:
    body = {"jsonrpc": "2.0", "id": 1, "method": method, "params": params}
    r = session.post(SOLANA_RPC_URL, data=json.dumps(body), timeout=15)
    r.raise_for_status()
    j = r.json()
    if "error" in j:
        raise RuntimeError(f"RPC error {method}: {j['error']}")
    return j.get("result")


def shannon(counter: Counter) -> float:
    total = sum(counter.values())
    if total <= 0:
        return 0.0
    h = 0.0
    for v in counter.values():
        if v <= 0:
            continue
        p = v / total
        h -= p * math.log2(p)
    # Normalize to [0,1] by dividing by log2(N) where N = unique categories
    n = len(counter)
    if n <= 1:
        return 0.0
    return h / math.log2(n)


def features_from_block(block: dict) -> tuple[list[float], dict]:
    """Compute the 9 Φ-features + per-vector data for a Solana block."""
    txs = block.get("transactions") or []
    fee_counter = Counter()
    cu_counter = Counter()        # log10-bucketed CU (used for f5)
    cu_linear_counter = Counter() # linear-bucketed CU (used for f8 — distinct from f5)
    program_counter = Counter()
    account_counter = Counter()
    instr_count_counter = Counter()
    err_counter = Counter()
    acct_per_tx_counter = Counter()   # for f7: accounts-per-tx complexity entropy
    joint_fee_cu_counter = Counter()  # for f9: joint fee × CU correlation entropy

    total_lamports = 0
    successful = 0
    failed = 0

    for entry in txs:
        meta = entry.get("meta") or {}
        tx = entry.get("transaction") or {}
        msg = (tx.get("message") if isinstance(tx, dict) else {}) or {}

        fee = int(meta.get("fee") or 0)
        cu = int(meta.get("computeUnitsConsumed") or 0)
        err = meta.get("err")

        # Bin lamport fees and CU into log buckets for entropy stability
        fee_bucket = 0 if fee == 0 else int(math.floor(math.log10(fee + 1)))
        cu_bucket = 0 if cu == 0 else int(math.floor(math.log10(cu + 1)))
        fee_counter[fee_bucket] += 1
        cu_counter[cu_bucket] += 1
        # Linear CU bucket for f8 (distinct from log10 cu_bucket used in f5)
        cu_lin_bucket = (0 if cu == 0 else
                         1 if cu < 50_000 else
                         2 if cu < 200_000 else
                         3 if cu < 500_000 else
                         4 if cu < 1_000_000 else 5)
        cu_linear_counter[cu_lin_bucket] += 1

        if err is None:
            successful += 1
            err_counter["ok"] += 1
        else:
            failed += 1
            err_counter["err"] += 1

        accts = msg.get("accountKeys") or []
        for a in accts:
            # accountKeys may be strings or {"pubkey": ...} objects
            key = a if isinstance(a, str) else a.get("pubkey") if isinstance(a, dict) else None
            if key:
                account_counter[key] += 1

        # f7 tracking: bin accounts-per-tx into log2 buckets (tx complexity diversity)
        n_accts = len(accts)
        acct_bin = 0 if n_accts == 0 else min(5, int(math.floor(math.log2(n_accts + 1))))
        acct_per_tx_counter[acct_bin] += 1

        # f9 tracking: joint (fee_bucket, cu_bucket) for correlation entropy
        joint_fee_cu_counter[(fee_bucket, cu_bucket)] += 1

        instrs = msg.get("instructions") or []
        instr_count_counter[len(instrs)] += 1
        for ix in instrs:
            pid_idx = ix.get("programIdIndex")
            if isinstance(pid_idx, int) and 0 <= pid_idx < len(accts):
                pid = accts[pid_idx] if isinstance(accts[pid_idx], str) else (accts[pid_idx].get("pubkey") if isinstance(accts[pid_idx], dict) else None)
                if pid:
                    program_counter[pid] += 1

        total_lamports += fee

    # f1..f9 per L1.1 (Solana-mapped):
    # f1 fee entropy, f2 account diversity, f3 tx-success ratio entropy,
    # f4 program diversity, f5 cu entropy, f6 instr-count entropy,
    # f7 program concentration (1-Gini-style proxy via top-k share),
    # f8 cu/fee correlation proxy, f9 reserved (set to overall block entropy).
    f1 = shannon(fee_counter)
    f2 = shannon(account_counter)
    f3 = shannon(err_counter)
    f4 = shannon(program_counter)
    f5 = shannon(cu_counter)
    f6 = shannon(instr_count_counter)
    # f7: Shannon entropy of accounts-per-tx distribution (tx complexity diversity)
    #     H(log2-binned account counts per tx) — distinct from f2 (unique accounts)
    f7 = shannon(acct_per_tx_counter)
    # f8: Shannon entropy of CU linear-bucket distribution (distinct from f5)
    #     f5 uses log10 buckets (scale-invariant); f8 uses linear buckets (scale-sensitive)
    #     This captures whether CU distribution is skewed toward specific cost tiers
    f8 = shannon(cu_linear_counter)
    # f9: Shannon entropy of joint (fee_bucket × cu_bucket) distribution
    #     Captures fee-compute correlation patterns — independent signal from f1/f5
    f9 = shannon(joint_fee_cu_counter)

    feats = [f1, f2, f3, f4, f5, f6, f7, f8, f9]

    stats = {
        "tx_count": len(txs),
        "successful": successful,
        "failed": failed,
        "total_fee_lamports": total_lamports,
        "unique_programs": len(program_counter),
        "unique_accounts": len(account_counter),
        "phi": sum(feats) / 9.0,
    }
    return feats, stats


def vector_from_features(feats: list[float], slot: int) -> list[float]:
    """Expand the 9 features into a deterministic 128-dim behavioural vector.
    Bands 0-15 = f1, 16-31 = f2, ... 128-143 = f9 (last 17 reserved for slot
    salt to prevent collapse to a single FAISS centroid).
    """
    out: list[float] = []
    band = DIM // 9
    for v in feats:
        out.extend([float(v)] * band)
    # pad to DIM with a deterministic slot-derived salt
    seed = hashlib.sha256(f"{slot}".encode()).digest()
    while len(out) < DIM:
        b = seed[(len(out) - DIM) % 32]
        out.append((b / 255.0) - 0.5)
    return out[:DIM]


def push_block(slot: int, feats: list[float], stats: dict) -> None:
    entity_id = f"solana:slot:{slot}"
    vector = vector_from_features(feats, slot)
    payload = {
        "vectors": [{
            "entity_id":   entity_id,
            "vector":      vector,
            "magnitude":   max(1.0, math.log10(stats["total_fee_lamports"] + 1)),
            "entropy":     stats["phi"],
            "block_num":   slot,
            "chain_id":    SOLANA_CHAIN_ID,
            "chain_label": SOLANA_LABEL,
            "vm_type":     "SVM",
        }],
        "block_num":      slot,
        "block_features": feats,
        "block_phi":      stats["phi"],
        "entity_id":      entity_id,
        "chain_id":       SOLANA_CHAIN_ID,
        "chain_label":    SOLANA_LABEL,
        "vm_type":        "SVM",
        "metadata": {
            "source":       "trion-svm",
            "tx_count":     stats["tx_count"],
            "successful":   stats["successful"],
            "failed":       stats["failed"],
            "total_fee":    stats["total_fee_lamports"],
            "programs":     stats["unique_programs"],
            "accounts":     stats["unique_accounts"],
        },
    }
    url = f"{FAISS_SERVICE_URL}/index/add_batch"
    # Circuit breaker: when FAISS has been failing, back off so we don't
    # pile more requests into its queue (which is the main cause of
    # cascading 15s timeouts). Returns True on success, False on failure.
    global _faiss_failures, _faiss_skip_until
    now_ts = time.monotonic()
    if now_ts < _faiss_skip_until:
        return False
    try:
        # Short timeout (4s) so a slow FAISS doesn't hold the indexer thread
        # for 15s per slot. If FAISS can't respond in 4s it's saturated and
        # retrying will only make it worse — let the circuit breaker kick in.
        r = session.post(url, data=json.dumps(payload), timeout=4)
        if r.status_code != 200:
            print(f"[svm][slot={slot}] FAISS POST {r.status_code}: {r.text[:200]}", file=sys.stderr)
            _faiss_failures += 1
        else:
            j = r.json()
            print(f"[svm][slot={slot}] txs={stats['tx_count']:>4} ok={stats['successful']:>4} "
                  f"fail={stats['failed']:>3} prog={stats['unique_programs']:>3} "
                  f"acct={stats['unique_accounts']:>4} Φ={stats['phi']:.4f} "
                  f"→ added={j.get('added',0)} idx_total={j.get('indexed_vectors','?')}")
            _faiss_failures = 0
            return True
    except requests.RequestException as e:
        print(f"[svm][slot={slot}] FAISS POST failed: {e}", file=sys.stderr)
        _faiss_failures += 1
    # On 3+ consecutive failures: open circuit for 30s, giving FAISS room
    # to drain its queue and serve oracle reads (mental/spiritual/etc).
    if _faiss_failures >= 3:
        _faiss_skip_until = time.monotonic() + 30.0
        print(f"[svm] FAISS circuit OPEN for 30s (consecutive failures={_faiss_failures})", file=sys.stderr)
        _faiss_failures = 0
    return False


# Circuit-breaker state — used by post_to_faiss above.
_faiss_failures: int = 0
_faiss_skip_until: float = 0.0


def main() -> int:
    print("===============================================================")
    print(" TRION L0-SVM: SOLANA BEHAVIORAL INDEXER")
    print("===============================================================")
    print(f" RPC          : {SOLANA_RPC_URL}")
    print(f" Chain id     : {SOLANA_CHAIN_ID}  ({SOLANA_LABEL})")
    print(f" FAISS target : {FAISS_SERVICE_URL}")
    print(f" Poll         : {POLL_SLEEP_MS}ms")
    print(" Streaming live Solana slots into FAISS. Press Ctrl+C to stop.")
    print()

    # Public Solana RPCs typically only serve `getBlock` for slots that are
    # already FINALIZED. To stay reliably inside that window we keep our
    # cursor a fixed offset behind the finalized tip. SAFE_LAG = 40 slots
    # ≈ 16 s — well past the 32-block finality window.
    SAFE_LAG = 40
    last_slot = 0
    consecutive_failures = 0

    while True:
        try:
            current = rpc("getSlot", [{"commitment": "finalized"}])
            if not isinstance(current, int):
                time.sleep(POLL_SLEEP_MS / 1000.0)
                continue

            if last_slot == 0:
                last_slot = current - SAFE_LAG

            # Stay at least SAFE_LAG behind the finalized tip
            tip = current - SAFE_LAG
            if tip <= last_slot:
                time.sleep(POLL_SLEEP_MS / 1000.0)
                continue

            target = last_slot + 1
            try:
                block = rpc("getBlock", [target, {
                    "encoding": "json",
                    "transactionDetails": "full",
                    "maxSupportedTransactionVersion": 0,
                    "rewards": False,
                    "commitment": "finalized",
                }])
                consecutive_failures = 0
            except RuntimeError as e:
                msg = str(e)
                # Skipped slot (-32007 / -32009) — advance silently
                if "-32007" in msg or "-32009" in msg or "skipped" in msg.lower():
                    last_slot = target
                    continue
                # Slot data missing (-32004) — typically the public node hasn't
                # caught up. Skip ahead a bit and retry rather than spinning.
                if "-32004" in msg or "not available" in msg.lower() or "older than" in msg.lower():
                    consecutive_failures += 1
                    # Jump forward by min(consecutive_failures * 5, 100) slots,
                    # but never past the safe tip.
                    jump = min(consecutive_failures * 5, 100)
                    last_slot = min(last_slot + jump, tip)
                    if consecutive_failures % 20 == 1:
                        print(f"[svm] block missing for slot {target}, advancing +{jump} (cursor={last_slot}, tip={tip})", file=sys.stderr)
                    time.sleep(0.3)
                    continue
                raise

            if not block:
                last_slot = target
                continue

            feats, stats = features_from_block(block)
            push_block(target, feats, stats)
            last_slot = target
            consecutive_failures = 0

        except KeyboardInterrupt:
            print("\n[svm] shutdown requested")
            return 0
        except requests.HTTPError as e:
            # 429 = rate limited. Public Solana RPCs throttle aggressively;
            # back off for a longer window and resume from current confirmed slot.
            status = getattr(e.response, "status_code", None)
            if status == 429:
                print(f"[svm] rate-limited (429), sleeping 15s", file=sys.stderr)
                time.sleep(15.0)
                last_slot = 0   # re-bootstrap from current confirmed slot
            else:
                print(f"[svm] HTTP {status}: {e}", file=sys.stderr)
                time.sleep(3.0)
        except Exception as e:
            msg = str(e)
            if "429" in msg or "Too Many Requests" in msg:
                print(f"[svm] rate-limited (429), sleeping 15s", file=sys.stderr)
                time.sleep(15.0)
                last_slot = 0
            else:
                print(f"[svm] loop error: {e}", file=sys.stderr)
                time.sleep(2.0)


if __name__ == "__main__":
    sys.exit(main())

;; gate.fc — TRION TON Execution Gate (TVM / FunC)
;;
;; Mirrors TRIONExecutionGate.sol on TON.
;; Behavioral firewall: gate_check passes when phi >= threshold.
;;
;; Storage layout:
;;   owner_addr (267 bits)
;;   gates_dict (dict, key=256-bit gate_id, value=gate cell ref)
;;
;; Gate cell:
;;   custom_threshold [64 bits] (0 = use route threshold)
;;   check_count      [64 bits]
;;   pass_count       [64 bits]
;;   block_count      [64 bits]
;;   last_phi         [64 bits]
;;
;; Op codes:
;;   0x01  set_gate_threshold
;;   0x02  gate_check  (owner/validator)
;;   0x10  transfer_ownership

#include "stdlib.fc";

const int ERR_UNAUTHORIZED  = 100;
const int ERR_INVALID_SCORE = 101;
const int ERR_GATE_BLOCKED  = 120;
const int ERR_GATE_NOT_FOUND = 121;


;; ── Storage ───────────────────────────────────────────────────────────────────

(slice, cell) load_storage() inline {
    slice ds = get_data().begin_parse();
    slice owner_addr = ds~load_bits(267);
    cell gates_dict  = ds~load_ref();
    return (owner_addr, gates_dict);
}

() save_storage(slice owner_addr, cell gates_dict) impure inline {
    set_data(begin_cell()
        .store_slice(owner_addr)
        .store_ref(gates_dict)
    .end_cell());
}

cell pack_gate(int threshold, int checks, int passes, int blocks, int phi) inline {
    return begin_cell()
        .store_uint(threshold, 64)
        .store_uint(checks,    64)
        .store_uint(passes,    64)
        .store_uint(blocks,    64)
        .store_uint(phi,       64)
    .end_cell();
}

(int, int, int, int, int) unpack_gate(cell c) inline {
    slice s = c.begin_parse();
    int threshold = s~load_uint(64);
    int checks    = s~load_uint(64);
    int passes    = s~load_uint(64);
    int blocks    = s~load_uint(64);
    int phi       = s~load_uint(64);
    return (threshold, checks, passes, blocks, phi);
}

;; ── recv_internal ─────────────────────────────────────────────────────────────

() recv_internal(int msg_value, cell in_msg_full, slice in_msg_body) impure {
    if (in_msg_body.slice_empty?()) { return (); }

    slice cs = in_msg_full.begin_parse();
    cs~load_uint(4);
    slice sender_addr = cs~load_bits(267);

    int op = in_msg_body~load_uint(32);

    ;; ── 0x01 set_gate_threshold ──────────────────────────────────────────────
    if (op == 1) {
        (slice owner_addr, cell gates_dict) = load_storage();
        throw_unless(ERR_UNAUTHORIZED, equal_slices(sender_addr, owner_addr));

        int gate_id   = in_msg_body~load_uint(256);
        int threshold = in_msg_body~load_uint(64);
        throw_if(ERR_INVALID_SCORE, threshold > 1000000);

        slice gkey = begin_cell().store_uint(gate_id, 256).end_cell().begin_parse();
        (slice existing, int found) = gates_dict.dict_get?(256, gkey);

        int checks = 0; int passes = 0; int blocks = 0; int phi = 0;
        if (found) {
            cell gc = existing~load_ref();
            (_, checks, passes, blocks, phi) = unpack_gate(gc);
        }

        cell new_gc = pack_gate(threshold, checks, passes, blocks, phi);
        gates_dict~dict_set(256, gkey, new_gc.begin_parse());
        save_storage(owner_addr, gates_dict);
        return ();
    }

    ;; ── 0x02 gate_check ──────────────────────────────────────────────────────
    if (op == 2) {
        (slice owner_addr, cell gates_dict) = load_storage();

        int gate_id        = in_msg_body~load_uint(256);
        int phi            = in_msg_body~load_uint(64);
        int route_threshold = in_msg_body~load_uint(64);

        slice gkey = begin_cell().store_uint(gate_id, 256).end_cell().begin_parse();
        (slice existing, int found) = gates_dict.dict_get?(256, gkey);

        int threshold = route_threshold;
        int checks = 0; int passes = 0; int blocks = 0;

        if (found) {
            cell gc = existing~load_ref();
            (int ct, int ch, int pa, int bl, _) = unpack_gate(gc);
            if (ct > 0) { threshold = ct; }
            checks = ch; passes = pa; blocks = bl;
        }

        checks = checks + 1;
        int passed = (phi >= threshold) ? 1 : 0;
        if (passed) { passes = passes + 1; }
        else { blocks = blocks + 1; }

        cell new_gc = pack_gate(threshold, checks, passes, blocks, phi);
        gates_dict~dict_set(256, gkey, new_gc.begin_parse());
        save_storage(owner_addr, gates_dict);

        throw_unless(ERR_GATE_BLOCKED, passed);
        return ();
    }

    ;; ── 0x10 transfer_ownership ──────────────────────────────────────────────
    if (op == 16) {
        (slice owner_addr, cell gates_dict) = load_storage();
        throw_unless(ERR_UNAUTHORIZED, equal_slices(sender_addr, owner_addr));
        slice new_owner = in_msg_body~load_bits(267);
        save_storage(new_owner, gates_dict);
        return ();
    }
}

;; ── get_methods ───────────────────────────────────────────────────────────────

(int, int, int, int) get_gate_stats(int gate_id) method_id {
    (_, cell gates_dict) = load_storage();
    slice gkey = begin_cell().store_uint(gate_id, 256).end_cell().begin_parse();
    (slice existing, int found) = gates_dict.dict_get?(256, gkey);
    throw_unless(ERR_GATE_NOT_FOUND, found);
    cell gc = existing~load_ref();
    (int threshold, int checks, int passes, int blocks, int phi) = unpack_gate(gc);
    return (threshold, checks, passes, blocks);
}

;; intent.fc — TRION TON Intent Registry (TVM / FunC)
;;
;; Mirrors BTCPIntent.sol on TON.
;; Intent actions: 0=SWAP 1=TRANSFER 2=LIQUIDITY 3=STAKE 4=BORROW
;; Intent statuses: 0=PENDING 1=ROUTING 2=EXECUTING 3=COMPLETED 4=FAILED 5=EXPIRED 6=RESURRECTED

const int ERR_INTENT_NOT_FOUND       = 111;

;; ── Status transition validator ───────────────────────────────────────────────

int valid_transition(int from, int to) inline {
    ;; PENDING→ROUTING|FAILED|EXPIRED  ROUTING→EXECUTING|FAILED|EXPIRED
    ;; EXECUTING→COMPLETED|FAILED  FAILED→RESURRECTED
    if ((from == 0) & ((to == 1) | (to == 4) | (to == 5))) { return 1; }
    if ((from == 1) & ((to == 2) | (to == 4) | (to == 5))) { return 1; }
    if ((from == 2) & ((to == 3) | (to == 4)))              { return 1; }
    if ((from == 4) & (to == 6))                            { return 1; }
    return 0;
}

;; ── recv_internal ─────────────────────────────────────────────────────────────

() recv_internal(int msg_value, cell in_msg_full, slice in_msg_body) impure {
    if (in_msg_body.slice_empty?()) { return (); }

    slice cs = in_msg_full.begin_parse();
    cs~load_uint(4);
    slice sender_addr = cs~load_bits(267);

    int op = in_msg_body~load_uint(32);

    ;; ── 0x01 register_intent ─────────────────────────────────────────────────
    if (op == 1) {
        (slice owner_addr, int intent_count, cell intents_dict) = load_storage();

        int hash   = in_msg_body~load_uint(256);
        int entity = in_msg_body~load_uint(256);
        int action = in_msg_body~load_uint(8);
        int ain    = in_msg_body~load_uint(256);
        int aout   = in_msg_body~load_uint(256);
        int mag    = in_msg_body~load_uint(64);
        int chain  = in_msg_body~load_uint(64);
        int dl     = in_msg_body~load_uint(64);

        throw_if(ERR_INVALID_ACTION, action > 4);
        throw_if(ERR_ZERO_MAGNITUDE, mag == 0);

        slice key = begin_cell().store_uint(hash, 256).end_cell().begin_parse();
        (_, int found) = intents_dict.dict_get?(256, key);
        throw_if(ERR_INTENT_EXISTS, found);

        int ts = now();
        cell ic = pack_intent(hash, entity, action, ain, aout, mag, chain, dl, 0, ts);
        slice val = ic.begin_parse();
        intents_dict~dict_set(256, key, val);
        save_storage(owner_addr, intent_count + 1, intents_dict);
        return ();
    }

    ;; ── 0x02 update_intent_status ────────────────────────────────────────────
    if (op == 2) {
        (slice owner_addr, int intent_count, cell intents_dict) = load_storage();
        throw_unless(ERR_UNAUTHORIZED, equal_slices(sender_addr, owner_addr));

        int hash       = in_msg_body~load_uint(256);
        int new_status = in_msg_body~load_uint(8);

        slice key = begin_cell().store_uint(hash, 256).end_cell().begin_parse();
        (slice rec_slice, int found) = intents_dict.dict_get?(256, key);
        throw_unless(ERR_INTENT_NOT_FOUND, found);

        ;; read current status from the cell
        cell rec_cell = rec_slice~load_ref();
        slice rs = rec_cell.begin_parse();
        rs~skip_bits(256 + 256 + 8 + 256 + 256 + 64 + 64 + 64); ;; skip to status
        int cur_status = rs~load_uint(8);

        throw_unless(ERR_INVALID_TRANSITION, valid_transition(cur_status, new_status));

        ;; re-read full record to rebuild
        slice rs2 = rec_cell.begin_parse();
        int h  = rs2~load_uint(256); int e  = rs2~load_uint(256);
        int a  = rs2~load_uint(8);   int ai = rs2~load_uint(256);
        int ao = rs2~load_uint(256); int m  = rs2~load_uint(64);
        int sc = rs2~load_uint(64);  int d  = rs2~load_uint(64);
        rs2~load_uint(8); ;; old status
        int ts = rs2~load_uint(64);

        cell new_cell = pack_intent(h, e, a, ai, ao, m, sc, d, new_status, ts);
        slice new_val = new_cell.begin_parse();
        intents_dict~dict_set(256, key, new_val);
        save_storage(owner_addr, intent_count, intents_dict);
        return ();
    }
}

        .store_uint(settled_at,    64)
        .store_uint(revert_at,     64)
        .store_uint(revert_code,    8)
    .end_cell();
}

;; ── recv_internal ─────────────────────────────────────────────────────────────

() recv_internal(int msg_value, cell in_msg_full, slice in_msg_body) impure {
    if (in_msg_body.slice_empty?()) { return (); }

    slice cs = in_msg_full.begin_parse();
    cs~load_uint(4);
    slice sender_addr = cs~load_bits(267);

    int op = in_msg_body~load_uint(32);

    ;; ── 0x01 commit_liquidity ────────────────────────────────────────────────
    if (op == 1) {
        (slice owner_addr, int ccount, cell cdict) = load_storage();

        int cid       = in_msg_body~load_uint(256);
        int entity    = in_msg_body~load_uint(256);
        int route     = in_msg_body~load_uint(256);
        int asset     = in_msg_body~load_uint(256);
        int amount    = in_msg_body~load_uint(64);
        int min_coh   = in_msg_body~load_uint(64);
        int expiry    = in_msg_body~load_uint(64);

        throw_if(ERR_INVALID_SCORE, min_coh > 1000000);

        slice ckey = begin_cell().store_uint(cid, 256).end_cell().begin_parse();
        (_, int found) = cdict.dict_get?(256, ckey);
        throw_if(ERR_COMMITMENT_EXISTS, found);

        int ts = now();
        throw_if(ERR_EXPIRED, expiry <= ts);

        cell cc = pack_commitment(entity, route, asset, amount, min_coh, expiry, 0, ts, 0, 0, 0, 0);
        cdict~dict_set(256, ckey, cc.begin_parse());
        save_storage(owner_addr, ccount + 1, cdict);
        return ();
    }

    ;; ── 0x02 settle_commitment ───────────────────────────────────────────────
    if (op == 2) {
        (slice owner_addr, int ccount, cell cdict) = load_storage();
        throw_unless(ERR_UNAUTHORIZED, equal_slices(sender_addr, owner_addr));

        int cid      = in_msg_body~load_uint(256);
        int exec_bh  = in_msg_body~load_uint(256);
        int coherence = in_msg_body~load_uint(64);

        slice ckey = begin_cell().store_uint(cid, 256).end_cell().begin_parse();
        (slice cv, int found) = cdict.dict_get?(256, ckey);
        throw_unless(ERR_COMMITMENT_MISSING, found);

        cell cc = cv~load_ref();
        slice s = cc.begin_parse();
        int entity  = s~load_uint(256);
        int route   = s~load_uint(256);
        int asset   = s~load_uint(256);
        int amount  = s~load_uint(64);
        int min_coh = s~load_uint(64);
        int expiry  = s~load_uint(64);
        int status  = s~load_uint(8);
        int ts      = s~load_uint(64);

        throw_if(ERR_ALREADY_FINAL, status != 0);
        throw_if(ERR_EXPIRED, now() > expiry);
        throw_if(ERR_COHERENCE_LOW, coherence < min_coh);

        cell new_cc = pack_commitment(entity, route, asset, amount, min_coh, expiry, 1, ts, exec_bh, now(), 0, 0);
        cdict~dict_set(256, ckey, new_cc.begin_parse());
        save_storage(owner_addr, ccount, cdict);
        return ();
    }

    ;; ── 0x03 revert_commitment ───────────────────────────────────────────────
    if (op == 3) {
        (slice owner_addr, int ccount, cell cdict) = load_storage();
        throw_unless(ERR_UNAUTHORIZED, equal_slices(sender_addr, owner_addr));

        int cid    = in_msg_body~load_uint(256);
        int reason = in_msg_body~load_uint(8);

        slice ckey = begin_cell().store_uint(cid, 256).end_cell().begin_parse();
        (slice cv, int found) = cdict.dict_get?(256, ckey);
        throw_unless(ERR_COMMITMENT_MISSING, found);

        cell cc = cv~load_ref();
        slice s = cc.begin_parse();
        int entity  = s~load_uint(256);
        int route   = s~load_uint(256);
        int asset   = s~load_uint(256);
        int amount  = s~load_uint(64);
        int min_coh = s~load_uint(64);
        int expiry  = s~load_uint(64);
        int status  = s~load_uint(8);
        int ts      = s~load_uint(64);

        throw_if(ERR_ALREADY_FINAL, status != 0);

        cell new_cc = pack_commitment(entity, route, asset, amount, min_coh, expiry, 2, ts, 0, 0, now(), reason);
        cdict~dict_set(256, ckey, new_cc.begin_parse());
        save_storage(owner_addr, ccount, cdict);
        return ();
    }

    ;; ── 0x10 transfer_ownership ──────────────────────────────────────────────
    if (op == 16) {
        (slice owner_addr, int ccount, cell cdict) = load_storage();
        throw_unless(ERR_UNAUTHORIZED, equal_slices(sender_addr, owner_addr));
        slice new_owner = in_msg_body~load_bits(267);
        save_storage(new_owner, ccount, cdict);
        return ();
    }
}

;; ── get_methods ───────────────────────────────────────────────────────────────

(int, int, int) get_commitment(int cid) method_id {
    (_, _, cell cdict) = load_storage();
    slice ckey = begin_cell().store_uint(cid, 256).end_cell().begin_parse();
    (slice cv, int found) = cdict.dict_get?(256, ckey);
    throw_unless(ERR_COMMITMENT_MISSING, found);
    cell cc = cv~load_ref();
    slice s = cc.begin_parse();
    s~skip_bits(256 + 256 + 256); ;; entity, route, asset
    int amount  = s~load_uint(64);
    int min_coh = s~load_uint(64);
    s~skip_bits(64);              ;; expiry
    int status  = s~load_uint(8);
    return (amount, min_coh, status);
}

int get_commitment_count() method_id {
    (_, int c, _) = load_storage();
    return c;
}

;; oracle.fc — TRION TON Oracle (TVM / FunC)
;;
;; Equivalent to TRIONOracleV3.sol on TON.
;; Provides publish_btcp_route + verify_execution for cross-VM consensus.
;;
;; Storage layout (cell):
;;   owner_addr  (267 bits, MsgAddressInt)
;;   route_count (64 bits)
;;   version     (8 bits)
;;   routes_dict (dict, key = 256-bit route_id, value = route cell)
;;
;; Route cell layout:
;;   route_id        [256 bits]
;;   anchor_bh       [256 bits]
;;   execution_bh    [256 bits]
;;   coherence_score [64 bits]  (×1_000_000)
;;   threshold_score [64 bits]
;;   published_at    [64 bits]  (unix timestamp)
;;   is_active       [1 bit]
;;
;; Op codes:
;;   0x01  publish_btcp_route
;;   0x02  verify_execution
;;   0x03  add_validator
;;   0x10  transfer_ownership

#include "stdlib.fc";

;; ── Error codes ───────────────────────────────────────────────────────────────
const int ERR_UNAUTHORIZED    = 100;
const int ERR_INVALID_SCORE   = 101;
const int ERR_ROUTE_NOT_FOUND = 102;
const int ERR_ROUTE_INACTIVE  = 103;

;; ── Storage helpers ───────────────────────────────────────────────────────────

(slice, int, int, cell) load_storage() inline {
    slice ds = get_data().begin_parse();
    slice owner_addr  = ds~load_bits(267);
    int route_count   = ds~load_uint(64);
    int version       = ds~load_uint(8);
    cell routes_dict  = ds~load_ref();
    return (owner_addr, route_count, version, routes_dict);
}

() save_storage(slice owner_addr, int route_count, int version, cell routes_dict) impure inline {
    set_data(begin_cell()
        .store_slice(owner_addr)
        .store_uint(route_count, 64)
        .store_uint(version, 8)
        .store_ref(routes_dict)
    .end_cell());
}

;; ── Route serialisation ───────────────────────────────────────────────────────

cell pack_route(int route_id, int anchor_bh, int exec_bh, int coherence, int threshold, int ts, int active) inline {
    return begin_cell()
        .store_uint(route_id,  256)
        .store_uint(anchor_bh, 256)
        .store_uint(exec_bh,   256)
        .store_uint(coherence, 64)
        .store_uint(threshold, 64)
        .store_uint(ts,        64)
        .store_uint(active,    1)
    .end_cell();
}

(int, int, int, int, int, int, int) unpack_route(cell c) inline {
    slice s = c.begin_parse();
    int route_id  = s~load_uint(256);
    int anchor_bh = s~load_uint(256);
    int exec_bh   = s~load_uint(256);
    int coherence = s~load_uint(64);
    int threshold = s~load_uint(64);
    int ts        = s~load_uint(64);
    int active    = s~load_uint(1);
    return (route_id, anchor_bh, exec_bh, coherence, threshold, ts, active);
}

;; ── Auth helper ───────────────────────────────────────────────────────────────

() assert_owner(slice sender_addr) impure inline {
    (slice owner_addr, _, _, _) = load_storage();
    throw_unless(ERR_UNAUTHORIZED, equal_slices(sender_addr, owner_addr));
}


;; ── recv_internal ─────────────────────────────────────────────────────────────

() recv_internal(int msg_value, cell in_msg_full, slice in_msg_body) impure {
    if (in_msg_body.slice_empty?()) { return (); }

    slice cs = in_msg_full.begin_parse();
    cs~load_uint(4);              ;; flags
    slice sender_addr = cs~load_bits(267);

    int op = in_msg_body~load_uint(32);

    ;; ── 0x01 publish_btcp_route ───────────────────────────────────────────────
    if (op == 1) {
        (slice owner_addr, int route_count, int version, cell routes_dict) = load_storage();
        throw_unless(ERR_UNAUTHORIZED, equal_slices(sender_addr, owner_addr));

        int route_id  = in_msg_body~load_uint(256);
        int anchor_bh = in_msg_body~load_uint(256);
        int exec_bh   = in_msg_body~load_uint(256);
        int coherence = in_msg_body~load_uint(64);
        int threshold = in_msg_body~load_uint(64);

        throw_if(ERR_INVALID_SCORE, coherence > 1000000);
        throw_if(ERR_INVALID_SCORE, threshold > 1000000);

        int ts = now();
        cell route_cell = pack_route(route_id, anchor_bh, exec_bh, coherence, threshold, ts, 1);

        ;; store route in dict keyed by route_id (256-bit)
        slice key = begin_cell().store_uint(route_id, 256).end_cell().begin_parse();
        slice val = route_cell.begin_parse();
        routes_dict~dict_set(256, key, val);

        save_storage(owner_addr, route_count + 1, version, routes_dict);
        return ();
    }

    ;; ── 0x02 verify_execution — returns (is_safe, coherence, threshold) ───────
    if (op == 2) {
        (_, _, _, cell routes_dict) = load_storage();
        int route_id = in_msg_body~load_uint(256);

        slice key = begin_cell().store_uint(route_id, 256).end_cell().begin_parse();
        (slice route_slice, int found) = routes_dict.dict_get?(256, key);
        throw_unless(ERR_ROUTE_NOT_FOUND, found);

        cell route_cell = route_slice~load_ref();
        (_, _, _, int coherence, int threshold, _, int active) = unpack_route(route_cell);
        throw_unless(ERR_ROUTE_INACTIVE, active);

        ;; Emit log: is_safe coherence threshold
        ;; (TON uses out-messages for events; minimal log here)
        return ();
    }

    ;; ── 0x03 add_validator (no-op storage; validators managed off-chain) ──────
    if (op == 3) {
        assert_owner(sender_addr);
        return ();
    }

    ;; ── 0x10 transfer_ownership ──────────────────────────────────────────────
    if (op == 16) {
        (_, int route_count, int version, cell routes_dict) = load_storage();
        assert_owner(sender_addr);
        slice new_owner = in_msg_body~load_bits(267);
        save_storage(new_owner, route_count, version, routes_dict);
        return ();
    }
}

;; ── get_methods ───────────────────────────────────────────────────────────────

(int, int, int) get_route_info(int route_id) method_id {
    (_, _, _, cell routes_dict) = load_storage();
    slice key = begin_cell().store_uint(route_id, 256).end_cell().begin_parse();
    (slice route_slice, int found) = routes_dict.dict_get?(256, key);
    throw_unless(ERR_ROUTE_NOT_FOUND, found);
    cell route_cell = route_slice~load_ref();
    (_, _, _, int coherence, int threshold, _, int active) = unpack_route(route_cell);
    return (coherence, threshold, active);
}

int get_route_count() method_id {
    (_, int route_count, _, _) = load_storage();
    return route_count;
}

int get_version() method_id {
    (_, _, int version, _) = load_storage();
    return version;
}

slice get_owner() method_id {
    (slice owner_addr, _, _, _) = load_storage();
    return owner_addr;
}
