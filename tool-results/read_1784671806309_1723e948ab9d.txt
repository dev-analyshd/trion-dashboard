    86→function MiniSparkline({ data, color, width = 64, height = 24 }: { data: number[]; color: string; width?: number; height?: number }) {
    87→  const min = Math.min(...data);
    88→  const max = Math.max(...data);
    89→  const range = max - min || 1;
    90→  const pts = data
    91→    .map((v, i) => {
    92→      const x = (i / (data.length - 1)) * width;
    93→      const y = height - ((v - min) / range) * (height - 4) - 2;
    94→      return `${x},${y}`;
    95→    })
    96→    .join(" ");
    97→  return (
    98→    <svg width={width} height={height} className="shrink-0">
    99→      <polyline points={pts} fill="none" stroke={color} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
   100→    </svg>
   101→  );
   102→}
