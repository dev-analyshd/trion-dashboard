   101→// ─── Mini Sparkline SVG ─────────────────────────────────────────
   102→function MiniSparkline({
   103→  data,
   104→  color = "#00D4AA",
   105→  width = 120,
   106→  height = 32,
   107→}: {
   108→  data: number[];
   109→  color?: string;
   110→  width?: number;
   111→  height?: number;
   112→}) {
   113→  if (data.length < 2) return null;
   114→  const min = Math.min(...data);
   115→  const max = Math.max(...data);
   116→  const range = max - min || 1;
   117→  const points = data
   118→    .map((v, i) => {
   119→      const x = (i / (data.length - 1)) * width;
   120→      const y = height - ((v - min) / range) * (height - 4) - 2;
   121→      return `${x},${y}`;
   122→    })
   123→    .join(" ");
   124→  return (
   125→    <svg width={width} height={height} className="overflow-visible">
   126→      <defs>
   127→        <linearGradient id={`grad-${color.slice(1)}`} x1="0" y1="0" x2="0" y2="1">
   128→          <stop offset="0%" stopColor={color} stopOpacity={0.3} />
   129→          <stop offset="100%" stopColor={color} stopOpacity={0} />
   130→        </linearGradient>
   131→      </defs>
   132→      <polygon
   133→        points={`0,${height} ${points} ${width},${height}`}
   134→        fill={`url(#grad-${color.slice(1)})`}
   135→      />
   136→      <polyline points={points} fill="none" stroke={color} strokeWidth="1.5" />
   137→    </svg>
   138→  );
   139→}
