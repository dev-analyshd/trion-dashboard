    55→    55→    55→function Sparkline({ data, color = "#10B981", width = 80, height = 28 }: { data: number[]; color?: string; width?: number; height?: number }) {
    56→    56→    56→  const min = Math.min(...data), max = Math.max(...data);
    57→    57→    57→  const range = max - min || 1;
    58→    58→    58→  const pts = data.map((v, i) => `${(i / (data.length - 1)) * width},${height - ((v - min) / range) * height}`).join(" ");
    59→    59→    59→  return (
    60→    60→    60→    <svg width={width} height={height} className="overflow-visible">
    61→    61→    61→      <defs>
    62→    62→    62→        <linearGradient id={`grad-${color.replace("#","")}`} x1="0" y1="0" x2="0" y2="1">
    63→    63→    63→          <stop offset="0%" stopColor={color} stopOpacity="0.15" />
    64→    64→    64→          <stop offset="100%" stopColor={color} stopOpacity="0" />
    65→    65→    65→        </linearGradient>
    66→    66→    66→      </defs>
    67→    67→    67→      <path d={`M0,${height} L${pts} L${width},${height} Z`} fill={`url(#grad-${color.replace("#","")})`} />
    68→    68→    68→      <polyline points={pts} fill="none" stroke={color} strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
    69→    69→    69→    </svg>
    70→    70→    70→  );
    71→    71→    71→}
