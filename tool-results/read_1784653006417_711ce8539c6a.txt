    45→// ── Signal encoding ───────────────────────────────────────────────────────────
    46→function buildSignalHash(signal) {
    47→  const sigId = (signal.signal_id || "").replace(/^0x/, "").padEnd(32, "0").slice(0, 32);
    48→  const ts    = Math.floor(Date.now() / 1000).toString(16).padStart(16, "0");
    49→  const coh   = Math.floor((signal.coherence || 0.5) * 1e6).toString(16).padStart(8, "0");
    50→  return "0x" + crypto.createHash("sha256").update(sigId + ts + coh).digest("hex");
    51→}
    52→
    53→function buildMemo(signal) {
    54→  const hash = buildSignalHash(signal);
    55→  return `TRION:${hash.slice(2, 18)}:c${Math.floor((signal.coherence || 0.5) * 1000)}`;
    56→}

   157→async function publishUtxo(chain, signal) {
   158→  const wif  = process.env[chain.envKey];
   159→  const memo = buildMemo(signal);
   160→
   161→  if (!wif) {
   162→    console.log(`  [${chain.key.toUpperCase()}] DRY_RUN — key not set`);
   163→    recordResult(chain.key, { mode: "DRY_RUN", memo, last_error: null });
   164→    return;
   165→  }
   166→
   167→  let bitcoin, ECPairFactory, ecc;
   168→  try {
   169→    bitcoin = require("bitcoinjs-lib");
   170→    ecc     = require("tiny-secp256k1");
   171→    const ecpair = require("ecpair");
   172→    ECPairFactory = ecpair.ECPairFactory || ecpair.default?.ECPairFactory || ecpair;
   173→  } catch (e) {
   174→    console.warn(`  [${chain.key.toUpperCase()}] bitcoinjs-lib not available: ${e.message}`);
   175→    recordResult(chain.key, { mode: "REAL", last_status: "sdk_missing", memo, last_error: e.message });
   176→    return;
   177→  }
   178→
   179→  try {
   180→    bitcoin.initEccLib(ecc);
   181→    const ECPair = typeof ECPairFactory === "function"
   182→      ? ECPairFactory(ecc)
   183→      : ECPairFactory.ECPairFactory(ecc);
   184→
   185→    const network = BTC_NETWORKS[chain.network] || bitcoin.networks.bitcoin;
   186→    const keyPair = ECPair.fromWIF(wif, network);
   187→
   188→    // Derive address (p2wpkh for BTC/LTC, p2pkh for DOGE/DASH)
   189→    let payment;
   190→    if (chain.addressType === "p2wpkh") {
   191→      payment = bitcoin.payments.p2wpkh({ pubkey: keyPair.publicKey, network });
   192→    } else {
   193→      payment = bitcoin.payments.p2pkh({ pubkey: keyPair.publicKey, network });
   194→    }
   195→    const address = payment.address;
   196→
   197→    // Fetch UTXOs
   198→    let utxos = [];
   199→    try {
   200→      const utxoResp = await axios.get(chain.utxoApi(address), { timeout: 8000 });
   201→      // Normalize different API response formats
   202→      if (chain.key === "btc") {
   203→        utxos = utxoResp.data || [];
   204→      } else {
   205→        // BlockCypher format (used for DOGE, LTC): txrefs array
   206→        utxos = utxoResp.data?.txrefs || utxoResp.data?.unspent_outputs || utxoResp.data?.txs || utxoResp.data?.utxos || utxoResp.data || [];
   207→      }
   208→    } catch (e) {
   209→      console.warn(`  [${chain.key.toUpperCase()}] UTXO fetch error: ${e.message}`);
   210→    }
   211→
   212→    if (!utxos.length) {
   213→      // No UTXOs — record a signed block proof
   214→      const proofData = `TRION_PROOF:${chain.key.toUpperCase()}:${address}:${memo}`;
   215→      const proofHash = crypto.createHash("sha256").update(proofData).digest("hex");
   216→      console.log(`  [${chain.key.toUpperCase()}] No UTXOs at ${address} — block proof=${proofHash.slice(0, 16)}`);
   217→      recordResult(chain.key, {
   218→        mode: "REAL", last_status: "block_proof", address, memo,
   219→        proof_hash: proofHash, last_error: "no_utxos",
   220→      });
   221→      return;
   222→    }
   223→
   224→    // Build OP_RETURN transaction
   225→    const opReturnData = Buffer.from(buildSignalHash(signal).slice(2, 34), "hex"); // 16 bytes
   226→    const opReturn     = bitcoin.script.compile([bitcoin.opcodes.OP_RETURN, opReturnData]);
   227→
   228→    // Use first UTXO
   229→    const utxo = utxos[0];
   230→    const txidNorm = utxo.txid || utxo.tx_hash;
   231→    const voutNorm = utxo.vout ?? utxo.tx_output_n ?? 0;
   232→    const valueNorm = utxo.value ?? utxo.amount ?? 0;
   233→
   234→    const psbt = new bitcoin.Psbt({ network });
   235→    if (chain.addressType === "p2wpkh") {
   236→      psbt.addInput({
   237→        hash: txidNorm, index: voutNorm,
   238→        witnessUtxo: { script: payment.output, value: valueNorm },
   239→      });
   240→    } else {
   241→      // Legacy P2PKH needs full txHex (simplification: use non-segwit input)
   242→      psbt.addInput({ hash: txidNorm, index: voutNorm });
   243→    }
   244→    psbt.addOutput({ script: opReturn, value: 0 });
   245→
   246→    psbt.signInput(0, keyPair);
   247→    psbt.finalizeAllInputs();
   248→    const txHex = psbt.extractTransaction().toHex();
   249→
   250→    // Broadcast
   251→    let broadcastResp;
   252→    if (chain.key === "btc") {
   253→      broadcastResp = await axios.post(chain.broadcastUrl, txHex,
   254→        { headers: { "Content-Type": "text/plain" }, timeout: 10000 });
   255→      const txid = broadcastResp.data;
   256→      console.log(`  [${chain.key.toUpperCase()}] txid=${txid}`);
   257→      recordResult(chain.key, { mode: "LIVE", txid, address, memo, last_error: null });
   258→    } else {
   259→      broadcastResp = await axios.post(chain.broadcastUrl, { rawtx: txHex }, { timeout: 10000 });
   260→      const txid = broadcastResp.data?.tx?.hash || broadcastResp.data?.txid || broadcastResp.data;
   261→      console.log(`  [${chain.key.toUpperCase()}] txid=${txid}`);
   262→      recordResult(chain.key, { mode: "LIVE", txid, address, memo, last_error: null });
   263→    }
   264→  } catch (e) {
   265→    const errMsg = e?.response?.data?.error || e?.response?.data || e.message;
   266→    console.warn(`  [${chain.key.toUpperCase()}] error: ${errMsg}`);
   267→    recordResult(chain.key, { mode: "REAL", last_status: "error", memo, last_error: String(errMsg).slice(0, 120) });
   268→  }
   269→}

   316→async function publishCosmos(chain, signal) {
   317→  const hexKey = process.env[chain.envKey];
   318→  const memo   = buildMemo(signal);
   319→
   320→  if (!hexKey) {
   321→    console.log(`  [${chain.name}] DRY_RUN — key not set`);
   322→    recordResult(chain.key, { mode: "DRY_RUN", memo, last_error: null });
   323→    return;
   324→  }
   325→
   326→  try {
   327→    const { DirectSecp256k1Wallet }   = await import("@cosmjs/proto-signing");
   328→    const { SigningStargateClient, GasPrice } = await import("@cosmjs/stargate");
   329→
   330→    const keyBytes = Uint8Array.from(Buffer.from(hexKey.replace(/^0x/, ""), "hex"));
   331→    const wallet   = await DirectSecp256k1Wallet.fromKey(keyBytes, chain.prefix);
   332→    const [{ address }] = await wallet.getAccounts();
   333→
   334→    const client = await SigningStargateClient.connectWithSigner(
   335→      chain.rpcUrl, wallet,
   336→      { gasPrice: GasPrice.fromString(`0.025${chain.denom}`) }
   337→    );
   338→
   339→    // Self-transfer of 1 minimal unit with signal memo
   340→    const result = await client.sendTokens(
   341→      address, address,
   342→      [{ denom: chain.denom, amount: "1" }],
   343→      "auto", memo
   344→    );
   345→
   346→    if (result.code === 0) {
   347→      console.log(`  [${chain.name}] txHash=${result.transactionHash} addr=${address}`);
   348→      recordResult(chain.key, {
   349→        mode: "LIVE", txHash: result.transactionHash,
   350→        address, memo, chain_id: chain.chainId, last_error: null,
   351→      });
   352→    } else {
   353→      throw new Error(`tx code=${result.code} log=${result.rawLog}`);
   354→    }
   355→  } catch (e) {
   356→    const msg = e.message || String(e);
   357→    console.warn(`  [${chain.name}] ${msg.slice(0, 100)} — block proof`);
   358→    recordResult(chain.key, {
   359→      mode: "REAL", last_status: "block_proof", memo,
   360→      chain_id: chain.chainId, last_error: msg.slice(0, 120),
   361→    });
   362→  }
   363→}

   377→async function publishMove(chain, signal) {
   378→  const rawKey = process.env[chain.envKey];
   379→  const memo   = buildMemo(signal);
   380→
   381→  if (!rawKey) {
   382→    console.log(`  [${chain.name}] DRY_RUN — key not set`);
   383→    recordResult(chain.key, { mode: "DRY_RUN", memo, last_error: null });
   384→    return;
   385→  }
   386→
   387→  try {
   388→    const {
   389→      Aptos, AptosConfig, Network, Account, Ed25519PrivateKey,
   390→    } = await import("@aptos-labs/ts-sdk");
   391→
   392→    // Use AIP-80 formatted key directly to suppress the deprecation warning.
   393→    // AIP-80 format: "ed25519-priv-0x<hex>". If the env var already carries the
   394→    // prefix we pass it through unchanged; otherwise we normalise to AIP-80.
   395→    const hexKey     = rawKey.replace(/^ed25519-priv-/, "").replace(/^0x/, "");
   396→    const aip80Key   = `ed25519-priv-0x${hexKey}`;
   397→    const privateKey = new Ed25519PrivateKey(aip80Key);
   398→    const account    = Account.fromPrivateKey({ privateKey });
   399→
   400→    const config = new AptosConfig({
   401→      network: chain.isMovement ? Network.CUSTOM : Network.MAINNET,
   402→      fullnode: chain.apiUrl,
   403→    });
   404→    const aptos = new Aptos(config);
   405→
   406→    // Build a simple self-transfer (0 APT) to anchor the signal on-chain
   407→    // Amount 0 is valid and costs only gas
   408→    const txn = await aptos.transaction.build.simple({
   409→      sender: account.accountAddress,
   410→      data: {
   411→        function: "0x1::aptos_account::transfer",
   412→        functionArguments: [account.accountAddress, 0n],
   413→      },
   414→    });
   415→
   416→    const committed = await aptos.signAndSubmitTransaction({ signer: account, transaction: txn });
   417→    const executed  = await aptos.waitForTransaction({ transactionHash: committed.hash });
   418→
   419→    if (executed.success) {
   420→      console.log(`  [${chain.name}] txHash=${committed.hash}`);
   421→      recordResult(chain.key, {
   422→        mode: "LIVE", txHash: committed.hash,
   423→        address: account.accountAddress.toString(), memo, last_error: null,
   424→      });
   425→    } else {
   426→      throw new Error(`tx failed vm_status=${executed.vm_status}`);
   427→    }
   428→  } catch (e) {
   429→    const msg = e.message || String(e);
   430→    console.warn(`  [${chain.name}] ${msg.slice(0, 100)} — block proof`);
   431→    recordResult(chain.key, {
   432→      mode: "REAL", last_status: "block_proof", memo, last_error: msg.slice(0, 120),
   433→    });
   434→  }
   435→}

   439→// SDK: @mysten/sui — programmable transaction with coin transfer to self

   442→async function publishSui(signal) {
   443→  const rawKey = process.env.SUI_PRIVATE_KEY;
   444→  const memo   = buildMemo(signal);
   445→
   446→  if (!rawKey) {
   447→    console.log(`  [SUI] DRY_RUN — key not set`);
   448→    recordResult("sui", { mode: "DRY_RUN", memo, last_error: null });
   449→    return;
   450→  }
   451→
   452→  try {
   453→    const { SuiClient, getFullnodeUrl } = await import("@mysten/sui/client");
   454→    const { Ed25519Keypair }            = await import("@mysten/sui/keypairs/ed25519");
   455→    const { Transaction }               = await import("@mysten/sui/transactions");
   456→
   457→    const keypair = Ed25519Keypair.fromSecretKey(rawKey);
   458→    const client  = new SuiClient({ url: getFullnodeUrl("mainnet") });
   459→    const address = keypair.getPublicKey().toSuiAddress();
   460→
   461→    // Check for gas coins
   462→    const { data: coins } = await client.getCoins({ owner: address, limit: 1 });
   463→
   464→    if (!coins || coins.length === 0) {
   465→      const checkpoint = await client.getLatestCheckpointSequenceNumber();
   466→      console.warn(`  [SUI] No SUI for gas at ${address} — block proof checkpoint=${checkpoint}`);
   467→      recordResult("sui", {
   468→        mode: "REAL", last_status: "block_proof", address, memo,
   469→        checkpoint: checkpoint.toString(), last_error: "no_gas_coins",
   470→      });
   471→      return;
   472→    }
   473→
   474→    // Build a programmable transaction — split 0 MIST from gas and merge back
   475→    const tx = new Transaction();
   476→    tx.setSender(address);
   477→    const [coin] = tx.splitCoins(tx.gas, [0]);
   478→    tx.mergeCoins(tx.gas, [coin]);
   479→    tx.setGasBudget(3_000_000);
   480→
   481→    const result = await client.signAndExecuteTransaction({
   482→      signer: keypair,
   483→      transaction: tx,
   484→      options: { showEffects: true },
   485→    });
   486→
   487→    if (result.effects?.status?.status === "success") {
   488→      console.log(`  [SUI] txDigest=${result.digest}`);
   489→      recordResult("sui", {
   490→        mode: "LIVE", txDigest: result.digest, address, memo, last_error: null,
   491→      });
   492→    } else {
   493→      throw new Error(`SUI tx failed: ${result.effects?.status?.error}`);
   494→    }
   495→  } catch (e) {
   496→    const msg = e.message || String(e);
   497→    console.warn(`  [SUI] ${msg.slice(0, 100)} — block proof`);
   498→    recordResult("sui", {
   499→      mode: "REAL", last_status: "block_proof", memo, last_error: msg.slice(0, 120),
   500→    });
   501→  }
   502→}

   516→async function publishTron(signal) {
   517→  const hexKey = process.env.TRON_PRIVATE_KEY;
   518→  const memo   = buildMemo(signal);
   519→
   520→  if (!hexKey) {
   521→    console.log(`  [TRON] DRY_RUN — key not set`);
   522→    recordResult("tron", { mode: "DRY_RUN", memo, last_error: null });
   523→    return;
   524→  }
   525→
   526→  try {
   527→    const tronApiKey = process.env.TRON_API_KEY || "";
   528→    const headers    = tronApiKey ? { "TRON-PRO-API-KEY": tronApiKey } : {};
   529→    const ownerAddr  = tronAddressFromPrivateKey(hexKey.replace(/^0x/, ""));
   530→
   531→    // Step 1: Create an unsigned transaction (self-transfer 1 SUN)
   532→    const createResp = await axios.post(
   533→      "https://api.trongrid.io/wallet/createtransaction",
   534→      { owner_address: ownerAddr, to_address: ownerAddr, amount: 1 },
   535→      { timeout: 10000, headers }
   536→    );
   537→
   538→    const txData = createResp.data;
   539→    if (!txData.raw_data_hex) {
   540→      throw new Error(`TronGrid error: ${JSON.stringify(txData).slice(0, 100)}`);
   541→    }
   542→
   543→    // Step 2: Sign — TRON signing = secp256k1.sign(sha256(raw_data_bytes), privkey)
   544→    const signingKey    = new ethers.SigningKey("0x" + hexKey.replace(/^0x/, ""));
   545→    const rawDataBytes  = Buffer.from(txData.raw_data_hex, "hex");
   546→    const rawDataDigest = ethers.sha256(rawDataBytes); // 0x-prefixed 32-byte hash
   547→    const sig           = signingKey.sign(rawDataDigest);
   548→
   549→    // TRON expects r + s + v where v is 0 or 1 (not Ethereum's 27/28)
   550→    const v      = sig.v - 27;
   551→    const sigHex = sig.r.slice(2) + sig.s.slice(2) + v.toString(16).padStart(2, "0");
   552→
   553→    // Step 3: Broadcast
   554→    const broadcastResp = await axios.post(
   555→      "https://api.trongrid.io/wallet/broadcasttransaction",
   556→      { ...txData, signature: [sigHex] },
   557→      { timeout: 10000, headers }
   558→    );
   559→
   560→    if (broadcastResp.data?.result === true) {
   561→      const txId = txData.txID;
   562→      console.log(`  [TRON] txId=${txId}`);
   563→      recordResult("tron", { mode: "LIVE", txId, address: ownerAddr, memo, last_error: null });
   564→    } else {
   565→      const errMsg = broadcastResp.data?.message || JSON.stringify(broadcastResp.data);
   566→      throw new Error(`Broadcast failed: ${errMsg}`);
   567→    }
   568→  } catch (e) {
   569→    const msg = e?.response?.data?.message || e.message || String(e);
   570→    console.warn(`  [TRON] ${msg.slice(0, 100)} — block proof`);
   571→    recordResult("tron", {
   572→      mode: "REAL", last_status: "block_proof", memo, last_error: msg.slice(0, 120),
   573→    });
   574→  }
   575→}

   579→// SDK: stellar-sdk — payment to self with text memo (28-char signal hash)

   582→async function publishPi(signal) {
   583→  const secretKey = process.env.PI_SECRET_KEY;
   584→  const memo      = buildMemo(signal).slice(0, 28); // Stellar text memo max 28 bytes
   585→
   586→  if (!secretKey) {
   587→    console.log(`  [PI] DRY_RUN — key not set`);
   588→    recordResult("pi", { mode: "DRY_RUN", memo, last_error: null });
   589→    return;
   590→  }
   591→
   592→  try {
   593→    const Stellar = await import("stellar-sdk");
   594→
   595→    const keypair = Stellar.Keypair.fromSecret(secretKey);
   596→
   597→    // Try Pi mainnet then Stellar public network as fallback
   598→    const endpoints = [
   599→      { url: "https://api.mainnet.minepi.com",  passphrase: "Pi Mainnet" },
   600→      { url: "https://horizon.stellar.org",      passphrase: Stellar.Networks.PUBLIC },
   601→    ];
   602→
   603→    for (const ep of endpoints) {
   604→      try {
   605→        const server  = new Stellar.Horizon.Server(ep.url, { allowHttp: false });
   606→        const account = await server.loadAccount(keypair.publicKey());
   607→        const baseFee = await server.fetchBaseFee();
   608→
   609→        const tx = new Stellar.TransactionBuilder(account, {
   610→          fee: String(baseFee),
   611→          networkPassphrase: ep.passphrase,
   612→        })
   613→          .addOperation(Stellar.Operation.payment({
   614→            destination: keypair.publicKey(),
   615→            asset:       Stellar.Asset.native(),
   616→            amount:      "0.0000001",
   617→          }))
   618→          .addMemo(Stellar.Memo.text(memo))
   619→          .setTimeout(30)
   620→          .build();
   621→
   622→        tx.sign(keypair);
   623→        const result = await server.submitTransaction(tx);
   624→
   625→        console.log(`  [PI] txHash=${result.hash} network=${ep.passphrase}`);
   626→        recordResult("pi", {
   627→          mode: "LIVE", txHash: result.hash,
   628→          address: keypair.publicKey(), memo, network: ep.passphrase, last_error: null,
   629→        });
   630→        return; // success — stop trying endpoints
   631→      } catch (epErr) {
   632→        console.warn(`  [PI] ${ep.url} — ${epErr.message}`);
   633→        continue;
   634→      }
   635→    }
   636→    throw new Error("All Pi/Stellar endpoints failed");
   637→  } catch (e) {
   638→    const msg = e.message || String(e);
   639→    console.warn(`  [PI] ${msg.slice(0, 100)} — block proof`);
   640→    recordResult("pi", {
   641→      mode: "REAL", last_status: "block_proof", memo, last_error: msg.slice(0, 120),
   642→    });
   643→  }

  1342→// ── Reflexive self-halt ────────────────────────────────────────────────────
  1343→// If TRION's own self-verification reports SILENCED, the protocol does not
  1344→// trust its own signal generation right now — skip broadcasting entirely.
  1345→async function checkSelfHalt() {
  1346→  try {
  1347→    const r = await axios.get(`${ORACLE_API_URL}/api/v1/self`, { timeout: 8000 });
  1348→    if (r.data?.status === "SILENCED") {
  1349→      console.warn(`[EXT-RELAYER][SELF-HALT] TRION self-coherence=${r.data.coherence} < threshold=${r.data.threshold} (limiting=${r.data.limiting_plane}) — skipping this cycle`);
  1350→      return true;
  1351→    }
  1352→    return false;
  1353→  } catch (e) {
  1354→    console.warn(`[EXT-RELAYER][SELF-HALT] could not reach /api/v1/self (${e.message}) — proceeding without self-halt check`);
  1355→    return false;
  1356→  }

  1358→
  1359→async function runCycle() {
  1360→  if (await checkSelfHalt()) {
  1361→    return;
  1362→  }
  1363→
  1364→  const entity  = MONITORED[0];
  1365→  const signal  = await fetchSignal(entity);
  1366→  const sigType = signal.signal_type ?? "BOOTSTRAP";
  1367→  const coh     = (signal.coherence ?? 0.5).toFixed(4);
  1368→
  1369→  console.log(`\n[EXT-RELAYER] ${new Date().toISOString()} — entity=${entity} signal=${sigType} coherence=${coh}`);
  1370→
  1371→  await Promise.allSettled([
  1372→    // ── UTXO (5) ──────────────────────────────────────────────────────────
  1373→    ...UTXO_CHAINS.map(c  => publishUtxo(c, signal)),
  1374→    publishCardano(signal),
  1375→    // ── Cosmos-SDK (11) ───────────────────────────────────────────────────
  1376→    ...COSMOS_CHAINS.map(c => publishCosmos(c, signal)),
  1377→    // ── Move VM (2) ───────────────────────────────────────────────────────
