   119→  const colors: Record<string, string> = {
   120→    GET: "rgba(0, 200, 220, 0.15)",

   126→    GET: "#00C8DC",
   127→    POST: "#00D4AA",
   128→    PUT: "#FFD93D",
   129→    DELETE: "#FF6B6B",
   130→  };
   131→  return (
   132→    <Badge
   133→      className="font-mono text-xs px-2 py-0.5 border-0"
   134→      style={{
   135→        backgroundColor: colors[method] || "rgba(255,255,255,0.1)",

   138→    >
