   138→// ─── Signal Generator ────────────────────────────────────────────
   139→function generateSignals(count: number): SignalEntry[] {
   140→  const entities = [
   141→    "0x7a25...3f4e", "0xd8dA...6045", "0x1f98...1fFe", "0xC02a...Ae66",
   142→    "0x6B17...1d0E", "0xA0b8...6eC2", "0x2260...FAC5", "0x5149...dC5e",
   143→    "0xBe00...48F0", "GXCk...NEAR", "EQC...TON", "0x4eF2...cE1A",
   144→    "0xDead...Beef", "0xSui...4a2b", "0x1a1a...2b2b",
   145→  ];
   146→  const chains = CHAINS.slice(0, 15);
   147→  const signalTypes = ["VALUATION", "SILENCE", "MANIPULATION_ALERT", "GENESIS", "TRAJECTORY_ANOMALY", "RESURRECTION", "FORK_DIVERGENCE", "LIQUIDITY_HEALTH", "CROSS_CHAIN_COHERENCE"];
   148→  const statuses: SignalEntry["status"][] = ["COHERENT", "COHERENT", "COHERENT", "COHERENT", "COHERENT", "SILENCE", "MANIPULATION_ALERT", "GENESIS", "COHERENT", "COHERENT", "CROSS_CHAIN_COHERENCE"];
   149→  const planes = ["Physical", "Mental", "Spiritual", "Conscious", "ANIMA"];
   150→  const now = new Date();
   151→  return Array.from({ length: count }, (_, i) => {
   152→    const t = new Date(now.getTime() - i * (1500 + Math.random() * 2500));
   153→    const chain = chains[Math.floor(Math.random() * chains.length)];
   154→    const sigType = signalTypes[Math.floor(Math.random() * signalTypes.length)];
   155→    const status = statuses[Math.floor(Math.random() * statuses.length)];
   156→    const coherence = status === "SILENCE" ? 0.28 + Math.random() * 0.2 : status === "MANIPULATION_ALERT" ? 0.4 + Math.random() * 0.15 : 0.68 + Math.random() * 0.28;
   157→    const threshold = 0.55 + Math.random() * 0.15;
   158→    return {
   159→      id: `SIG-${(100000 - i).toString().padStart(6, "0")}`,
   160→      time: t.toTimeString().split(" ")[0],
   161→      entity: entities[Math.floor(Math.random() * entities.length)],
   162→      entityShort: entities[Math.floor(Math.random() * entities.length)].slice(0, 8),
   163→      chain: chain.name, chainColor: chain.color,
   164→      signalType: sigType,
   165→      coherence: Math.round(coherence * 1000) / 1000,
   166→      threshold: Math.round(threshold * 1000) / 1000,
   167→      phi: Math.round((coherence * 0.25 + Math.random() * 0.1) * 1000) / 1000,
   168→      sigma: Math.round((coherence * 0.25 + Math.random() * 0.05) * 1000) / 1000,
   169→      anima: Math.round((0.1 + Math.random() * 0.6) * 1000) / 1000,
   170→      mental: Math.round((coherence * 0.3 + Math.random() * 0.1) * 1000) / 1000,
   171→      status, mfScore: Math.round(Math.random() * 100) / 100,
   172→      limitingPlane: planes[Math.floor(Math.random() * planes.length)],
   173→      dataSource: (i < 5) ? "LIVE" as const : "MOCK" as const,
   174→    };
   175→  });
   176→}
