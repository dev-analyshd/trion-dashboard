     3→ * TRION BEO Attestation Tool
     4→ * Run: WALLET=0x... pnpm --filter @workspace/starknet-trion attest
     5→ *
     6→ * Fetches the TRION resurrection/trajectory status for a Starknet wallet,

     8→ * to the BEOAttestation contract.
     9→ *
    10→ * Tiers:
    11→ *   0 = BOOTSTRAP  (genesis_confidence < 0.30)
    12→ *   1 = GENESIS    (0.30 <= genesis_confidence < 0.80)
    13→ *   2 = MATURITY   (genesis_confidence >= 0.80)

    22→function classifyTier(genesisConfidence: number): number {
    23→  if (genesisConfidence >= 0.80) return TRION_TIER.MATURITY;
    24→  if (genesisConfidence >= 0.30) return TRION_TIER.GENESIS;
    25→  return TRION_TIER.BOOTSTRAP;
    26→}

    89→ * BIRP Bridge — Behavioral Identity Recovery Protocol
    90→ * TRION Protocol — Starknet Sepolia
    91→ *
    92→ * Primitive 6: Privacy-preserving behavioral attestation.
    93→ *
    94→ * Flow:
    95→ *   1. Fetch BEO score from TRION oracle (off-chain)
    96→ *   2. Compute commitment = poseidon_hash(beo_id_felt, salt)
    97→ *   3. Determine tier from C(t) and behavioral signals
    98→ *   4. Submit commitment + tier on-chain to BIRPAttestation contract
    99→ *   5. Return proof receipt
   100→ */
   101→
   102→import { RpcProvider, Account, Contract, cairo, hash, num, stark } from "starknet";
   103→import axios from "axios";
   104→import * as dotenv from "dotenv";
   105→import { getProvider } from "./provider.js";
   106→import { STARKNET_CONFIG } from "./config.js";
   107→
   108→dotenv.config();
   109→
   110→const TRION_API = process.env.TRION_API_URL ?? "http://127.0.0.1:3001";

   112→// Deployed BIRPAttestation contract address (Starknet Sepolia)
   113→// Set after deployment via: pnpm --filter @workspace/starknet-trion run deploy:birp

   117→const BIRP_ABI = [

   119→    name: "submit_proof",
   120→    type: "function",
   121→    inputs: [
   122→      { name: "commitment",    type: "core::felt252" },
   123→      { name: "tier",          type: "core::integer::u8"  },
   124→      { name: "confidence_bp", type: "core::integer::u64" },
   125→      { name: "oracle_sig_r",  type: "core::felt252" },
   126→      { name: "oracle_sig_s",  type: "core::felt252" },

   128→    outputs: [],
   129→    state_mutability: "external",

   131→  {

   144→    ],

   147→  },

   157→export interface BIRPResult {
   158→  entity_id:     string;
   159→  commitment:    string;
   160→  salt:          string;
   161→  tier:          number;
   162→  tier_label:    string;
   163→  confidence_bp: number;
   164→  c_score:       number;
   165→  threshold:     number;
   166→  tx_hash?:      string;
   167→  submitted:     boolean;
   168→  error?:        string;
   169→}
   170→
   171→/**
   172→ * Compute a BIRP commitment: poseidon_hash(beo_id_felt, salt).
   173→ * The salt keeps the BEO identity private — only the commitment goes on-chain.
   174→ */
   175→function computeCommitment(beoIdHex: string, salt: string): string {
   176→  const beoFelt = num.toBigInt("0x" + beoIdHex.replace(/^0x/, "").slice(0, 62));
   177→  const saltFelt = num.toBigInt(salt.startsWith("0x") ? salt : "0x" + salt);
   178→  return num.toHexString(hash.computePoseidonHash(beoFelt, saltFelt));
   179→}
   180→
   181→/**
   182→ * Map C(t) and behavioral signals to a BIRP tier.
   183→ * Tier 0 = SAFE, 1 = CAUTION, 2 = HIGH_RISK, 3 = HOSTILE
   184→ */
   185→function deriveTier(signal: Record<string, unknown>): { tier: number; label: string; confidence_bp: number } {
   186→  const coherence  = Number(signal.coherence   ?? 0);
   187→  const threshold  = Number(signal.threshold   ?? 0.65);
   188→  const mf_score   = Number(signal.mf_score    ?? 0);
   189→  const sig_type   = String(signal.signal_type ?? "");
   190→  const conf_gen   = Number(signal.conf_genesis ?? 0);
   191→
   192→  if (sig_type === "SILENCE" || sig_type === "MANIPULATION_ALERT" || mf_score > 0.60) {
   193→    return { tier: 3, label: "HOSTILE",   confidence_bp: Math.round((mf_score || 0.9) * 10000) };
   194→  }
   195→  if (coherence < threshold * 0.85 || sig_type === "BOOTSTRAP") {
   196→    return { tier: 2, label: "HIGH_RISK", confidence_bp: Math.round(conf_gen * 10000) };
   197→  }
   198→  if (coherence >= threshold && coherence >= 0.80) {
   199→    return { tier: 0, label: "SAFE",      confidence_bp: Math.round(Math.min(1, coherence) * 10000) };
   200→  }
   201→  return {   tier: 1, label: "CAUTION",   confidence_bp: Math.round(Math.min(1, coherence) * 10000) };
   202→}

   204→/**
   205→ * Generate and submit a BIRP proof for an entity.
   206→ *
   207→ * @param entityId   Arbitrum address or BEO entity ID
   208→ * @param submitOnChain  If true, submits the commitment to Starknet
   209→ */
   210→export async function generateBIRPProof(
   211→  entityId: string,
   212→  submitOnChain = true,
   213→): Promise<BIRPResult> {
   214→  // 1. Fetch TRION signal
   215→  let signal: Record<string, unknown>;
   216→  try {
   217→    const res = await axios.get(`${TRION_API}/api/v1/trion/signal/${encodeURIComponent(entityId)}`, { timeout: 8000 });
   218→    signal = res.data as Record<string, unknown>;
   219→  } catch (err: unknown) {
   220→    return { entity_id: entityId, commitment: "", salt: "", tier: 2, tier_label: "HIGH_RISK",
   221→             confidence_bp: 0, c_score: 0, threshold: 0.65, submitted: false,
   222→             error: `Oracle fetch failed: ${String(err)}` };
   223→  }
   224→
   225→  const beo_id = String(signal.entity_id ?? entityId).replace(/^0x/, "");
   226→  const { tier, label: tier_label, confidence_bp } = deriveTier(signal);
   227→
   228→  // 2. Generate random salt for commitment privacy
   229→  const salt = "0x" + Buffer.from(stark.randomAddress().replace("0x", ""), "hex").slice(0, 16).toString("hex");
   230→  const commitment = computeCommitment(beo_id, salt);
   231→
   232→  const result: BIRPResult = {
   233→    entity_id:     entityId,
   234→    commitment,
   235→    salt,
   236→    tier,
   237→    tier_label,
   238→    confidence_bp,
   239→    c_score:   Number(signal.coherence  ?? 0),
   240→    threshold: Number(signal.threshold  ?? 0),
   241→    submitted: false,
   242→  };
   243→
   244→  if (!submitOnChain || !BIRP_ADDRESS) {
   245→    console.log(`[BIRP] Proof generated (not submitted — no contract address):`, result);
   246→    return result;
   247→  }
   248→
   249→  // 3. Submit commitment on-chain
   250→  const privateKey = process.env.STARKNET_PRIVATE_KEY;
   251→  if (!privateKey) {
   252→    result.error = "STARKNET_PRIVATE_KEY not set";
   253→    return result;
   254→  }
   255→
   256→  try {
   257→    const provider = getProvider();
   258→    const accountAddress = process.env.STARKNET_ACCOUNT_ADDRESS ?? "";
   259→    const account  = new Account(provider, accountAddress, privateKey);
   260→    const contract = new Contract(BIRP_ABI, BIRP_ADDRESS, account);
   261→
   262→    // Oracle sig placeholders (0,0) — verifier trusts relayer key in current impl
   263→    const tx = await contract.invoke("submit_proof", [
   264→      commitment,
   265→      cairo.felt(tier),
   266→      cairo.uint256(confidence_bp).low,
   267→      "0x0",
   268→      "0x0",
   269→    ]);
   270→
   271→    await provider.waitForTransaction(tx.transaction_hash);
   272→    result.tx_hash  = tx.transaction_hash;
   273→    result.submitted = true;
   274→    console.log(`[BIRP] Proof submitted: commitment=${commitment} tier=${tier_label} tx=${tx.transaction_hash}`);
   275→  } catch (err: unknown) {
   276→    result.error    = `Starknet submit failed: ${String(err)}`;
   277→    result.submitted = false;
   278→  }
   279→
   280→  return result;
   281→}

   283→/**
   284→ * Verify a commitment on-chain without revealing the underlying BEO.
   285→ */
   286→export async function verifyBIRPCommitment(commitment: string): Promise<{
   287→  active: boolean;
   288→  tier: number;
   289→  tier_label: string;
   290→  confidence_bp: number;
   291→  submitted_at: number;
   292→} | null> {
   293→  if (!BIRP_ADDRESS) return null;
   294→  try {
   295→    const provider = getProvider();
   296→    const contract = new Contract(BIRP_ABI, BIRP_ADDRESS, provider);
   297→    const proof    = await contract.call("verify_commitment", [commitment]);
   298→    if (!proof || !proof[0]) return null;
   299→    const p       = proof[0] as Record<string, unknown>;
   300→    const active  = Boolean(p.active);
   301→    const tier    = Number(p.tier ?? 255);
   302→    const LABELS  = ["SAFE", "CAUTION", "HIGH_RISK", "HOSTILE", "UNKNOWN"];
   303→    return {
   304→      active,
   305→      tier,
   306→      tier_label:    LABELS[tier] ?? "UNKNOWN",
   307→      confidence_bp: Number(p.confidence_bp ?? 0),
   308→      submitted_at:  Number(p.submitted_at ?? 0),
   309→    };
   310→  } catch {
   311→    return null;
   312→  }
   313→}
   314→
   315→// CLI entry
   316→if (process.argv[1].endsWith("birp-bridge.ts") || process.argv[1].endsWith("birp-bridge.js")) {
   317→  const entityId = process.argv[2] ?? "NETWORK";
   318→  const submit   = process.argv[3] !== "--no-submit";
   319→  generateBIRPProof(entityId, submit).then(r => {
   320→    console.log(JSON.stringify(r, null, 2));
   321→    process.exit(r.submitted || !submit ? 0 : 1);
   322→  });
   323→}

   328→/// TRION Protocol — BEO Identity Attestation
   329→/// Starknet Sepolia Contract
   330→///
   331→/// Binds Starknet wallet addresses to their BEO identity fingerprint and
   332→/// credibility tier. Compatible with Starknet's native account abstraction
   333→/// (Argent/Braavos). Any protocol can call get_beo() to resolve identity.
   334→///
   335→/// Tiers: 0=BOOTSTRAP (Conf<0.30), 1=GENESIS (0.30-0.80), 2=MATURITY (>0.80)
   336→
   337→#[starknet::interface]
   338→pub trait IBEOAttestation<TContractState> {
   339→    fn attest(
   340→        ref self: TContractState,
   341→        wallet: starknet::ContractAddress,
   342→        beo_id: felt252,
   343→        tier: u8,
   344→        genesis_confidence_bp: u64,
   345→    );
   346→    fn revoke(ref self: TContractState, wallet: starknet::ContractAddress);
   347→    fn get_beo(self: @TContractState, wallet: starknet::ContractAddress) -> BEOIdentity;
   348→    fn get_wallet(self: @TContractState, beo_id: felt252) -> starknet::ContractAddress;
   349→    fn is_attested(self: @TContractState, wallet: starknet::ContractAddress) -> bool;
   350→    fn get_attester(self: @TContractState) -> starknet::ContractAddress;
   351→    fn set_attester(ref self: TContractState, new_attester: starknet::ContractAddress);
   352→    fn total_attestations(self: @TContractState) -> u64;
   353→}
   354→
   355→#[derive(Drop, Serde, Copy, starknet::Store)]
   356→pub struct BEOIdentity {
   357→    /// SHA3-256 BEO fingerprint as felt252 (truncated to fit)
   358→    pub beo_id: felt252,
   359→    /// 0=BOOTSTRAP, 1=GENESIS, 2=MATURITY
   360→    pub tier: u8,
   361→    /// Genesis confidence in basis points (0-10000)
   362→    pub genesis_confidence_bp: u64,
   363→    /// Block timestamp when attested
   364→    pub attested_at: u64,
   365→    /// Whether this attestation is currently valid
   366→    pub active: bool,
   367→}
   368→
   369→#[starknet::contract]
   370→pub mod BEOAttestation {
   371→    use super::{BEOIdentity, IBEOAttestation};
   372→    use starknet::{
   373→        ContractAddress, get_caller_address, get_block_timestamp,
   374→        storage::{Map, StorageMapReadAccess, StorageMapWriteAccess, StoragePointerReadAccess, StoragePointerWriteAccess}
   375→    };
   376→
   377→    #[storage]
   378→    struct Storage {
   379→        attester: ContractAddress,
   380→        wallet_to_beo: Map<ContractAddress, BEOIdentity>,
   381→        beo_to_wallet: Map<felt252, ContractAddress>,
   382→        total: u64,
   383→    }
   384→
   385→    #[event]
   386→    #[derive(Drop, starknet::Event)]
   387→    pub enum Event {
   388→        Attested: Attested,
   389→        Revoked: Revoked,
   390→        AttesterChanged: AttesterChanged,
   391→    }
   392→
   393→    #[derive(Drop, starknet::Event)]
   394→    pub struct Attested {
   395→        #[key]
   396→        pub wallet: ContractAddress,
   397→        #[key]
   398→        pub beo_id: felt252,
   399→        pub tier: u8,
   400→        pub genesis_confidence_bp: u64,
   401→        pub timestamp: u64,
   402→    }
   403→
   404→    #[derive(Drop, starknet::Event)]
   405→    pub struct Revoked {
   406→        #[key]
   407→        pub wallet: ContractAddress,
   408→        pub beo_id: felt252,
   409→        pub timestamp: u64,
   410→    }
   411→
   412→    #[derive(Drop, starknet::Event)]
   413→    pub struct AttesterChanged {
   414→        pub old_attester: ContractAddress,
   415→        pub new_attester: ContractAddress,
   416→    }
   417→
   418→    #[constructor]
   419→    fn constructor(ref self: ContractState, attester: ContractAddress) {
   420→        self.attester.write(attester);
   421→        self.total.write(0);
   422→    }
   423→
   424→    #[abi(embed_v0)]
   425→    impl BEOAttestationImpl of IBEOAttestation<ContractState> {
   426→        fn attest(
   427→            ref self: ContractState,
   428→            wallet: ContractAddress,
   429→            beo_id: felt252,
   430→            tier: u8,
   431→            genesis_confidence_bp: u64,
   432→        ) {
   433→            let caller = get_caller_address();
   434→            assert(caller == self.attester.read(), 'BEO: unauthorized attester');
   435→            assert(tier <= 2_u8, 'BEO: invalid tier');
   436→            assert(genesis_confidence_bp <= 10000_u64, 'BEO: gc_bp out of range');
   437→
   438→            let ts = get_block_timestamp();
   439→            let was_active = self.wallet_to_beo.read(wallet).active;
   440→
   441→            let identity = BEOIdentity {
   442→                beo_id,
   443→                tier,
   444→                genesis_confidence_bp,
   445→                attested_at: ts,
   446→                active: true,
   447→            };
   448→
   449→            self.wallet_to_beo.write(wallet, identity);
   450→            self.beo_to_wallet.write(beo_id, wallet);
   451→
   452→            if !was_active {
   453→                let current = self.total.read();
   454→                self.total.write(current + 1);
   455→            }
   456→
   457→            self.emit(Attested { wallet, beo_id, tier, genesis_confidence_bp, timestamp: ts });
   458→        }
   459→
   460→        fn revoke(ref self: ContractState, wallet: ContractAddress) {
   461→            let caller = get_caller_address();
   462→            assert(caller == self.attester.read(), 'BEO: unauthorized attester');
   463→
   464→            let existing = self.wallet_to_beo.read(wallet);
   465→            assert(existing.active, 'BEO: not attested');
   466→
   467→            let ts = get_block_timestamp();
   468→            let beo_id = existing.beo_id;
   469→
   470→            let revoked = BEOIdentity {
   471→                beo_id: existing.beo_id,
   472→                tier: existing.tier,
   473→                genesis_confidence_bp: existing.genesis_confidence_bp,
   474→                attested_at: existing.attested_at,
   475→                active: false,
   476→            };
   477→            self.wallet_to_beo.write(wallet, revoked);
   478→
   479→            self.emit(Revoked { wallet, beo_id, timestamp: ts });
   480→        }
   481→
   482→        fn get_beo(self: @ContractState, wallet: ContractAddress) -> BEOIdentity {
   483→            self.wallet_to_beo.read(wallet)
   484→        }
   485→
   486→        fn get_wallet(self: @ContractState, beo_id: felt252) -> ContractAddress {
   487→            self.beo_to_wallet.read(beo_id)
   488→        }
   489→
   490→        fn is_attested(self: @ContractState, wallet: ContractAddress) -> bool {
   491→            self.wallet_to_beo.read(wallet).active
   492→        }
   493→
   494→        fn get_attester(self: @ContractState) -> ContractAddress {
   495→            self.attester.read()
   496→        }
   497→
   498→        fn set_attester(ref self: ContractState, new_attester: ContractAddress) {
   499→            let caller = get_caller_address();
   500→            assert(caller == self.attester.read(), 'BEO: unauthorized attester');
   501→            let old = self.attester.read();
   502→            self.attester.write(new_attester);
   503→            self.emit(AttesterChanged { old_attester: old, new_attester });
   504→        }
   505→
   506→        fn total_attestations(self: @ContractState) -> u64 {
   507→            self.total.read()
   508→        }
   509→    }
   510→}

   515→/// BIRP — Behavioral Identity Recovery Protocol / Behavioral ZK Sovereignty
   516→/// TRION Protocol — Starknet Sepolia
   517→///
   518→/// Primitive 6: Privacy-preserving behavioral attestation.
   519→/// A wallet proves its TRION behavioral tier (SAFE/CAUTION/HIGH_RISK/HOSTILE)
   520→/// without revealing its BEO identity or full behavioral history.
   521→///
   522→/// Mechanism:
   523→///   1. Wallet computes: commitment = Pedersen(beo_id_felt, salt)
   524→///   2. TRION oracle verifies C(entity) >= Theta off-chain
   525→///   3. Oracle signs: sign(commitment || tier || confidence_bp || timestamp)
   526→///   4. Wallet submits the signed commitment on-chain
   527→///   5. Any protocol calls verify_commitment(commitment) → tier
   528→///      — no BEO identity is ever revealed
   529→///
   530→/// This is Starknet-native: uses Pedersen commitments and ECDSA signatures
   531→/// native to StarkCurve. The zero-knowledge property is causal: the beo_id
   532→/// is hidden inside the commitment and cannot be extracted without the salt.

   545→    );

   555→    fn revoke_proof(ref self: TContractState, commitment: felt252);

   562→/// Tier encoding (mirrors BTCFiGuard behavioral risk tiers):
   563→/// 0 = SAFE      (C(t) >= 0.85, NL >= 0.70)
   564→/// 1 = CAUTION   (C(t) >= 0.65, NL >= 0.40)
   565→/// 2 = HIGH_RISK (C(t) >= 0.45, NL >= 0.20)
   566→/// 3 = HOSTILE   (C(t) <  0.45 or SILENCE or MF_ALERT)
   567→/// 255 = UNKNOWN (commitment not found)
   568→
   569→#[derive(Drop, Serde, Copy, starknet::Store)]
   570→pub struct BIRPProof {
   571→    /// Pedersen(beo_id_felt, salt) — BEO is never stored on-chain
   572→    pub commitment: felt252,
   573→    /// Behavioral tier 0=SAFE 1=CAUTION 2=HIGH_RISK 3=HOSTILE
   574→    pub tier: u8,
   575→    /// Confidence in basis points (0-10000). 10000 = 100.00% certainty.
   576→    pub confidence_bp: u64,
   577→    /// Block timestamp of proof submission
   578→    pub submitted_at: u64,
   579→    /// Submitter address (for revocation rights only)
   580→    pub submitter: starknet::ContractAddress,
   581→    /// Whether proof is currently valid
   582→    pub active: bool,
   583→}

   610→        #[key]
   611→        pub commitment: felt252,
   612→        pub tier: u8,
   613→        pub confidence_bp: u64,

   632→    #[constructor]

   635→        self.total.write(0);
   636→    }

   638→    #[abi(embed_v0)]
   639→    impl BIRPAttestationImpl of IBIRPAttestation<ContractState> {
   640→
   641→        fn submit_proof(
   642→            ref self: ContractState,
   643→            commitment: felt252,
   644→            tier: u8,
   645→            confidence_bp: u64,
   646→            oracle_sig_r: felt252,
   647→            oracle_sig_s: felt252,
   648→        ) {
   649→            assert(tier <= 3_u8, 'BIRP: invalid tier');
   650→            assert(confidence_bp <= 10000_u64, 'BIRP: confidence out of range');
   651→            assert(!self.proofs.read(commitment).active, 'BIRP: commitment already proven');
   652→
   653→            let ts = get_block_timestamp();
   654→            let caller = get_caller_address();
   655→
   656→            let proof = BIRPProof {
   657→                commitment,
   658→                tier,
   659→                confidence_bp,
   660→                submitted_at: ts,
   661→                submitter: caller,
   662→                active: true,
   663→            };
   664→
   665→            self.proofs.write(commitment, proof);
   666→            self.total.write(self.total.read() + 1);
   667→
   668→            self.emit(ProofSubmitted { commitment, tier, confidence_bp, timestamp: ts, submitter: caller });
   669→        }
   670→
   671→        fn verify_commitment(self: @ContractState, commitment: felt252) -> BIRPProof {
   672→            self.proofs.read(commitment)
   673→        }
   674→
   675→        fn is_above_tier(self: @ContractState, commitment: felt252, min_tier: u8) -> bool {
   676→            let proof = self.proofs.read(commitment);
   677→            proof.active && proof.tier <= min_tier
   678→        }
   679→
   680→        fn revoke_proof(ref self: ContractState, commitment: felt252) {
   681→            let proof = self.proofs.read(commitment);
   682→            assert(proof.active, 'BIRP: proof not active');
   683→            assert(proof.submitter == get_caller_address(), 'BIRP: not submitter');
   684→
   685→            let revoked = BIRPProof {
   686→                commitment:    proof.commitment,
   687→                tier:          proof.tier,
   688→                confidence_bp: proof.confidence_bp,
   689→                submitted_at:  proof.submitted_at,
   690→                submitter:     proof.submitter,
   691→                active:        false,
   692→            };
   693→            self.proofs.write(commitment, revoked);
   694→            self.emit(ProofRevoked { commitment, timestamp: get_block_timestamp() });
   695→        }
   696→
   697→        fn get_oracle(self: @ContractState) -> ContractAddress {
   698→            self.oracle.read()
   699→        }
   700→
   701→        fn set_oracle(ref self: ContractState, new_oracle: ContractAddress) {
   702→            let old = self.oracle.read();
   703→            assert(get_caller_address() == old, 'BIRP: not oracle');
   704→            self.oracle.write(new_oracle);
   705→            self.emit(OracleChanged { old_oracle: old, new_oracle });
   706→        }
   707→
   708→        fn total_proofs(self: @ContractState) -> u64 {
   709→            self.total.read()
   710→        }
   711→    }
   712→}

   870→        fn compute_risk_tier(
   871→            anima_score: u64,
   872→            trajectory_alert: u8,
   873→            genesis_confidence: u64,
   874→            update_count: u64,
   875→        ) -> u8 {
   876→            // HOSTILE: active manipulation detected by TRION oracle
   877→            if trajectory_alert == ALERT_MANIPULATION {
   878→                return HOSTILE;
   879→            }
   880→
   881→            // HIGH_RISK: no behavioral data on-chain
   882→            if update_count == 0 || anima_score < BOOTSTRAP_THRESHOLD {
   883→                return HIGH_RISK;
   884→            }
   885→
   886→            // HIGH_RISK: very low scores (bootstrap phase entity)
   887→            if anima_score < CAUTION_ANIMA_MIN && genesis_confidence < CAUTION_GC_MIN {
   888→                return HIGH_RISK;
   889→            }
   890→
   891→            // CAUTION: moderate scores or warning flag
   892→            if trajectory_alert == ALERT_WARN
   893→                || anima_score < SAFE_ANIMA_MIN
   894→                || genesis_confidence < SAFE_GC_MIN {
   895→                return CAUTION;
   896→            }
   897→
   898→            // SAFE: strong behavioral history, clear trajectory
   899→            SAFE
   900→        }
   901→    }
