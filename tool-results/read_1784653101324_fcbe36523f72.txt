   629→  1129→export async function getWorkingProvider(): Promise<RpcProvider> {
   630→  1130→  // Probe each endpoint with BOTH a chainId check AND a real getBlockWithTxs call.
   631→  1131→  // Some providers (e.g. drpc.org) advertise the right chainId but return
   632→  1132→  // -32601 "method does not exist" for starknet_getBlockWithTxs — that breaks
   633→  1133→  // the indexer hot loop. We must reject those endpoints up-front.
   634→  1134→  for (const url of STARKNET_CONFIG.rpcEndpoints) {
   635→  1135→    try {
   636→  1136→      const provider = new RpcProvider({ nodeUrl: url });
   637→  1137→      const chainId = await provider.getChainId();
   638→  1138→      if (chainId !== STARKNET_CONFIG.chainIdHex) continue;
   639→  1139→      // Probe getBlockWithTxs against latest block — must succeed.
   640→  1140→      const latest = await provider.getBlockNumber();
   641→  1141→      await provider.getBlockWithTxs(latest);
   642→  1142→      console.log(`✓ Connected to Starknet Mainnet via ${url}`);
   643→  1143→      console.log(`  Chain ID: ${chainId} (SN_MAIN), latest block: ${latest}`);
   644→  1144→      return provider;
   645→  1145→    } catch (e: any) {
   646→  1146→      const msg = e?.message ?? String(e);
   647→  1147→      console.log(`  · RPC ${url} rejected: ${msg.slice(0, 120)}`);
   648→  1148→    }
   649→  1149→  }
   650→  1150→  throw new Error('All Starknet Mainnet RPC endpoints failed (none support getBlockWithTxs)');
   651→  1151→}
